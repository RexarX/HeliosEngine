26[[nodiscard]] constexpr auto ResolveClientSize(
27 const ::helios::window::Properties& properties) noexcept
28 -> std::pair<int, int> {
29 const uint32_t width =
30 properties.client_width.value_or(properties.width.value_or(1U));
31 const uint32_t height =
32 properties.client_height.value_or(properties.height.value_or(1U));
33 return {static_cast<int>(width), static_cast<int>(height)};
