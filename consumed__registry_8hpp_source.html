28/**
29 * @brief Per-system registry for tracking consumed message ids.
30 * @details Each system running in parallel gets its own
31 * `ConsumedMessagesRegistry` instance. Systems mark messages as consumed by
32 * writing stable message ids into their registry, avoiding data races. At
33 * Update time, all registries are merged and applied to message queues in
34 * `MessageManager`.
35 *
36 * Entries are monotonic message ids assigned when messages enter the global
37 * manager — not transient indices into the previous/current buffers.

224inline auto ConsumedMessagesRegistry::EnsureIds(MessageTypeIndex type_index)
225 -> ConsumedIds& {
226 const auto it = consumed_.LowerBound(type_index);
227 if (it != consumed_.end() && it->first == type_index) {
228 return it->second;
229 }

232 .EmplaceHint(it, type_index, ConsumedIds{GetMemoryResource()})

236inline void ConsumedMessagesRegistry::MarkConsumed(MessageTypeIndex type_index,
237 AnyMessageId message_id) {
238 HELIOS_ASSERT(message_id.type == type_index,
239 "AnyMessageId type does not match type_index!");
240 auto& ids = EnsureIds(type_index);
241 // Insert in sorted order, maintaining uniqueness
242 const auto pos = std::ranges::lower_bound(ids, message_id);
243 if (pos == ids.end() || *pos != message_id) {
244 ids.insert(pos, message_id);
245 }
246}
