157constexpr void NativeWindows::Insert(ecs::Entity entity, NativeEntry entry) {
158 const auto it = FindEntry(entries, entity);
159 HELIOS_ASSERT(it == entries.end() || it->entity != entity,
160 "Native window entry already exists for entity '{}'!", entity);
