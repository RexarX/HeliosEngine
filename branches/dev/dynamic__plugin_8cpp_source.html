197auto DynamicPlugin::LoadPluginInstance() -> DynamicPluginResult<void> {
198 // Get create function
199 const auto create_result =
200 library_.GetSymbol<CreatePluginFn>(config_.create_symbol);
201 if (!create_result) [[unlikely]] {
202 log::Error("Create function '{}' not found in library!",
203 config_.create_symbol);
204 return std::unexpected(DynamicPluginError::kCreateSymbolNotFound);
205 }
206
207 // Get plugin type export function
208 const auto id_result =
209 library_.GetSymbol<PluginTypeExportFn>(config_.plugin_type_id_symbol);
210 if (!id_result) [[unlikely]] {
211 log::Error("Plugin ID function '{}' not found in library!",
212 config_.plugin_type_id_symbol);
213 return std::unexpected(DynamicPluginError::kIdSymbolNotFound);
214 }
215
216 const PluginTypeExportFn export_fn = *id_result;
217 const PluginTypeExport* export_info = export_fn();
218 if (export_info == nullptr) [[unlikely]] {
219 log::Error("Plugin type export function returned nullptr!");
220 return std::unexpected(DynamicPluginError::kCreateFailed);
221 }
222
223 const std::string_view qualified_name =
224 export_info->qualified_name != nullptr
225 ? std::string_view{export_info->qualified_name}
226 : std::string_view{};
227 plugin_type_id_ =
228 PluginTypeId::FromExported(export_info->hash, qualified_name);
229
230 // Create the plugin
231 CreatePluginFn create_fn = *create_result;
232 Plugin* raw_plugin = create_fn();
233
234 if (raw_plugin == nullptr) [[unlikely]] {
235 log::Error("Plugin creation function returned nullptr!");
236 return std::unexpected(DynamicPluginError::kCreateFailed);
237 }
238
239 plugin_.reset(raw_plugin);
240 return {};
241}
