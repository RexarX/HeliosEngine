74/**
75 * @brief Combines dirty flags.
76 * @param lhs Left-hand side flag
77 * @param rhs Right-hand side flag
78 * @return Combined flag
79 */
80[[nodiscard]] constexpr DirtyFlag operator|(DirtyFlag lhs,
81 DirtyFlag rhs) noexcept {
82 return static_cast<DirtyFlag>(std::to_underlying(lhs) |
83 std::to_underlying(rhs));
84}

86/**
87 * @brief Tests whether a dirty flag is set.
88 * @param flags Combined dirty flags
89 * @param flag Flag to test
90 * @return True if the flag is set, false otherwise
91 */
92[[nodiscard]] constexpr bool HasFlag(DirtyFlag flags, DirtyFlag flag) noexcept {
93 return (std::to_underlying(flags) & std::to_underlying(flag)) != 0U;
94}

214/**
215 * @brief Computes a default client size for a monitor resolution.
216 * @details Uses most of the screen on portrait displays and a balanced fraction
217 * on landscape displays so fixed 1280x720 is not assumed.
218 * @param screen_width Monitor width in pixels
219 * @param screen_height Monitor height in pixels
220 * @return Default width and height in pixels
221 */
222[[nodiscard]] constexpr auto DefaultSizeForScreen(
223 uint32_t screen_width, uint32_t screen_height) noexcept
224 -> std::pair<uint32_t, uint32_t> {
225 constexpr uint32_t kMinSize = 320;
226
227 if (screen_width == 0 || screen_height == 0) {
228 return {1280, 720};
229 }
230
231 if (screen_height > screen_width) {
232 const uint32_t width =
233 std::clamp((screen_width * 9U) / 10U, kMinSize, screen_width);
234 const uint32_t height =
235 std::clamp((width * 16U) / 10U, kMinSize, (screen_height * 9U) / 10U);
236 return {width, height};
237 }
238
239 const uint32_t width =
240 std::clamp((screen_width * 2U) / 3U, kMinSize, screen_width);
241 const uint32_t height =
242 std::clamp((screen_height * 2U) / 3U, kMinSize, screen_height);
243 return {width, height};
244}

363/**
364 * @brief Formats dirty flags as a pipe-separated list and writes to an output
365 * iterator.
366 * @tparam It Output iterator type
367 * @param out Output iterator to write the formatted string to
368 * @param flags Combined dirty flags
369 * @param with_prefix Whether to include a "DirtyFlag::" prefix for each flag
370 * @return Updated output iterator after writing the formatted string
371 */
372template <typename It>
373 requires std::output_iterator<It, char>
374inline It ToString(It out, DirtyFlag flags, bool with_prefix = false) {
375 const std::string_view kNoneStr = with_prefix ? "DirtyFlag::None" : "None";
376
377 if (flags == DirtyFlag::kNone) {
378 return std::format_to(out, "{}", kNoneStr);
379 }
380
381 bool first = true;
382 const auto append = [flags, &out, &first, with_prefix](std::string_view name,
383 DirtyFlag flag) {
384 if (!HasFlag(flags, flag)) {
385 return;
386 }
387 if (!first) {
388 out = std::format_to(out, " | ");
389 }
390 first = false;
391 out = with_prefix ? std::format_to(out, "DirtyFlag::{}", name)
392 : std::format_to(out, "{}", name);
393 };
394
395 append("Title", DirtyFlag::kTitle);
396 append("Size", DirtyFlag::kSize);
397 append("Pos", DirtyFlag::kPos);
398 append("Mode", DirtyFlag::kMode);
399 append("Cursor", DirtyFlag::kCursor);
400 append("Visible", DirtyFlag::kVisible);
401 append("Resizable", DirtyFlag::kResizable);
402 append("Decorated", DirtyFlag::kDecorated);
403 append("Monitor", DirtyFlag::kMonitor);
404 append("Icon", DirtyFlag::kIcon);
405 append("Maximized", DirtyFlag::kMaximized);
406 append("SizeLimits", DirtyFlag::kSizeLimits);
407 append("AspectRatio", DirtyFlag::kAspectRatio);
408 append("Opacity", DirtyFlag::kOpacity);
409 append("Floating", DirtyFlag::kFloating);
410 append("AutoIconify", DirtyFlag::kAutoIconify);
411 append("FocusOnShow", DirtyFlag::kFocusOnShow);
412 append("Attention", DirtyFlag::kAttention);
413 append("Focus", DirtyFlag::kFocus);
414 append("MousePassthrough", DirtyFlag::kMousePassthrough);
415 append("RefreshRate", DirtyFlag::kRefreshRate);
416 return out;
417}
