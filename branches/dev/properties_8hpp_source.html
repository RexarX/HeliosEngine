172 return (std::to_underlying(flags) & std::to_underlying(flag)) != 0U;

175/**
176 * @brief Formats dirty flags as a pipe-separated list and writes to an output
177 * iterator.
178 * @tparam It Output iterator type
179 * @param flags Combined dirty flags
180 * @param out Output iterator to write the formatted string to
181 * @param with_prefix Whether to include a "DirtyFlag::" prefix for each flag
182 * @return Updated output iterator after writing the formatted string
183 */
184template <typename It>
185 requires std::output_iterator<It, char>
186inline It ToString(DirtyFlag flags, It out, bool with_prefix = false) {
187 const std::string_view kNoneStr = with_prefix ? "DirtyFlag::None" : "None";
188
189 if (flags == DirtyFlag::kNone) {
190 return std::format_to(out, "{}", kNoneStr);
191 }
192
193 bool first = true;
194 const auto append = [flags, &out, &first, with_prefix](std::string_view name,
195 DirtyFlag flag) {
196 if (!HasFlag(flags, flag)) {
197 return;
198 }
199 if (!first) {
200 out = std::format_to(out, " | ");
201 }
202 first = false;
203 out = with_prefix ? std::format_to(out, "DirtyFlag::{}", name)
204 : std::format_to(out, "{}", name);
205 };
206
207 append("Title", DirtyFlag::kTitle);
208 append("Size", DirtyFlag::kSize);
209 append("Pos", DirtyFlag::kPos);
210 append("Mode", DirtyFlag::kMode);
211 append("Cursor", DirtyFlag::kCursor);
212 append("Visible", DirtyFlag::kVisible);
213 append("Resizable", DirtyFlag::kResizable);
214 append("Decorated", DirtyFlag::kDecorated);
215 append("Monitor", DirtyFlag::kMonitor);
216 append("Icon", DirtyFlag::kIcon);
217 append("Maximized", DirtyFlag::kMaximized);
218 append("SizeLimits", DirtyFlag::kSizeLimits);
219 append("AspectRatio", DirtyFlag::kAspectRatio);
220 append("Opacity", DirtyFlag::kOpacity);
221 append("Floating", DirtyFlag::kFloating);
222 append("AutoIconify", DirtyFlag::kAutoIconify);
223 append("FocusOnShow", DirtyFlag::kFocusOnShow);
224 append("Attention", DirtyFlag::kAttention);
225 append("Focus", DirtyFlag::kFocus);
226 append("MousePassthrough", DirtyFlag::kMousePassthrough);
227 append("RefreshRate", DirtyFlag::kRefreshRate);
228 return out;
229}

302/**
303 * @brief Computes a default client size for a monitor resolution.
304 * @details Uses most of the screen on portrait displays and a balanced fraction
305 * on landscape displays so fixed 1280x720 is not assumed.
306 * @param screen_width Monitor width in pixels
307 * @param screen_height Monitor height in pixels
308 * @return Default width and height in pixels
309 */
310[[nodiscard]] constexpr auto DefaultSizeForScreen(
311 uint32_t screen_width, uint32_t screen_height) noexcept
312 -> std::pair<uint32_t, uint32_t> {
313 constexpr uint32_t kMinSize = 320;
314
315 if (screen_width == 0 || screen_height == 0) {
316 return {1280, 720};
317 }
318
319 if (screen_height > screen_width) {
320 const uint32_t width =
321 std::clamp((screen_width * 9U) / 10U, kMinSize, screen_width);
322 const uint32_t height =
323 std::clamp((width * 16U) / 10U, kMinSize, (screen_height * 9U) / 10U);
324 return {width, height};
325 }
326
327 const uint32_t width =
328 std::clamp((screen_width * 2U) / 3U, kMinSize, screen_width);
329 const uint32_t height =
330 std::clamp((screen_height * 2U) / 3U, kMinSize, screen_height);
331 return {width, height};
332}
