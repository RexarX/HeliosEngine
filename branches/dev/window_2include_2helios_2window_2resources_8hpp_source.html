15/// @brief Conditions under which closing a window requests application exit.
16enum class ExitTrigger : uint8_t {
17 kNone = 0,
18 kPrimaryClosed = 1U << 0U, ///< A Primary-tagged window was destroyed.
19 kAllWindowsClosed = 1U << 1U, ///< The last native window was destroyed.
20 kAnyWindowClosed = 1U << 2U, ///< Any single window was destroyed.
21};

23/**
24 * @brief Combines exit triggers.
25 * @param lhs Left-hand side trigger flags
26 * @param rhs Right-hand side trigger flags
27 * @return Combined trigger flags
28 */
29[[nodiscard]] constexpr ExitTrigger operator|(ExitTrigger lhs,
30 ExitTrigger rhs) noexcept {
31 return static_cast<ExitTrigger>(std::to_underlying(lhs) |
32 std::to_underlying(rhs));
33}

35/**
36 * @brief Tests whether an exit trigger flag is set.
37 * @param triggers Combined exit trigger flags
38 * @param trigger Flag to test
39 * @return True if the flag is set, false otherwise
40 */
41[[nodiscard]] constexpr bool HasFlag(ExitTrigger triggers,
42 ExitTrigger trigger) noexcept {
43 return (std::to_underlying(triggers) & std::to_underlying(trigger)) != 0U;
44}

59/**
60 * @brief Determines whether a destroyed window should request application exit.
61 * @param triggers Configured exit triggers
62 * @param was_primary True if the destroyed window had the Primary component
63 * @param is_last_native_window True if no native windows remain after destroy
64 * @return True when an exit trigger matched
65 */
66[[nodiscard]] constexpr bool ShouldRequestExitOnClose(
67 ExitTrigger triggers, bool was_primary,
68 bool is_last_native_window) noexcept {
69 if (HasFlag(triggers, ExitTrigger::kAnyWindowClosed)) {
70 return true;
71 }
72 if (HasFlag(triggers, ExitTrigger::kPrimaryClosed) && was_primary) {
73 return true;
74 }
75 if (HasFlag(triggers, ExitTrigger::kAllWindowsClosed) &&
76 is_last_native_window) {
77 return true;
78 }
79 return false;
80}

82/**
83 * @brief Formats exit triggers as a pipe-separated list and writes to an output
84 * iterator.
85 * @tparam It Output iterator type
86 * @param triggers Combined exit trigger flags
87 * @param out Output iterator to write the formatted string to
88 * @param with_prefix Whether to include an "ExitTrigger::" prefix for each flag
89 * @return Updated output iterator after writing the formatted string
90 */
91template <typename It>
92 requires std::output_iterator<It, char>
93inline It ToString(ExitTrigger triggers, It out, bool with_prefix = false) {
94 const std::string_view kNoneStr = with_prefix ? "ExitTrigger::None" : "None";
95
96 if (triggers == ExitTrigger::kNone) {
97 return std::format_to(out, "{}", kNoneStr);
98 }
99
100 bool first = true;
101 const auto append = [triggers, &out, &first, with_prefix](
102 std::string_view name, ExitTrigger flag) {
103 if (!HasFlag(triggers, flag)) {
104 return;
105 }
106 if (!first) {
107 out = std::format_to(out, " | ");
108 }
109 first = false;
110 out = with_prefix ? std::format_to(out, "ExitTrigger::{}", name)
111 : std::format_to(out, "{}", name);
112 };
113
114 append("PrimaryClosed", ExitTrigger::kPrimaryClosed);
115 append("AllWindowsClosed", ExitTrigger::kAllWindowsClosed);
116 append("AnyWindowClosed", ExitTrigger::kAnyWindowClosed);
117 return out;
118}

120/**
121 * @brief Formats exit triggers as a pipe-separated list of flag names.
122 * @param triggers Combined exit trigger flags
123 * @param with_prefix Whether to include an "ExitTrigger::" prefix for each flag
124 * @return Formatted flag names
125 */
126[[nodiscard]] inline std::string ToString(ExitTrigger triggers,
127 bool with_prefix = false) {

129 result.reserve(64);
