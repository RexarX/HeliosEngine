16/// @brief Conditions under which closing a window requests application exit.
17enum class ExitTrigger : uint8_t {
18 kNone = 0,
19 kPrimaryClosed = 1U << 0U, ///< A Primary-tagged window was destroyed.
20 kAllWindowsClosed = 1U << 1U, ///< The last native window was destroyed.
21 kAnyWindowClosed = 1U << 2U, ///< Any single window was destroyed.
22};
23
24/**
25 * @brief Combines exit triggers.
26 * @param lhs Left-hand side trigger flags
27 * @param rhs Right-hand side trigger flags
28 * @return Combined trigger flags
29 */
30[[nodiscard]] constexpr ExitTrigger operator|(ExitTrigger lhs,
31 ExitTrigger rhs) noexcept {
32 return static_cast<ExitTrigger>(std::to_underlying(lhs) |
33 std::to_underlying(rhs));
34}
35
36/**
37 * @brief Tests whether an exit trigger flag is set.
38 * @param triggers Combined exit trigger flags
39 * @param trigger Flag to test
40 * @return True if the flag is set, false otherwise
41 */
42[[nodiscard]] constexpr bool HasFlag(ExitTrigger triggers,
43 ExitTrigger trigger) noexcept {
44 return (std::to_underlying(triggers) & std::to_underlying(trigger)) != 0U;
45}

47/// @brief Never auto-exit when windows close.
48inline constexpr auto kExitTriggersNone = ExitTrigger::kNone;
49
50/// @brief Exit when the last window is destroyed.
51inline constexpr auto kExitTriggersLastWindow = ExitTrigger::kAllWindowsClosed;
52
53/// @brief Exit when the primary window is destroyed.
54inline constexpr auto kExitTriggersPrimary = ExitTrigger::kPrimaryClosed;
55
56/// @brief Exit when the primary or last window is destroyed.
57inline constexpr auto kExitTriggersDefault =
58 ExitTrigger::kPrimaryClosed | ExitTrigger::kAllWindowsClosed;

60/**
61 * @brief Determines whether a destroyed window should request application exit.
62 * @param triggers Configured exit triggers
63 * @param was_primary True if the destroyed window had the Primary component
64 * @param is_last_native_window True if no native windows remain after destroy
65 * @return True when an exit trigger matched
66 */
67[[nodiscard]] constexpr bool ShouldRequestExitOnClose(
68 ExitTrigger triggers, bool was_primary,
69 bool is_last_native_window) noexcept {
70 if (HasFlag(triggers, ExitTrigger::kAnyWindowClosed)) {
71 return true;
72 }
73 if (HasFlag(triggers, ExitTrigger::kPrimaryClosed) && was_primary) {
74 return true;
75 }
76 if (HasFlag(triggers, ExitTrigger::kAllWindowsClosed) &&
77 is_last_native_window) {
78 return true;
79 }
80 return false;
81}

206/**
207 * @brief Formats exit triggers as a pipe-separated list and writes to an output
208 * iterator.
209 * @tparam It Output iterator type
210 * @param out Output iterator to write the formatted string to
211 * @param triggers Combined exit trigger flags
212 * @param with_prefix Whether to include an "ExitTrigger::" prefix for each flag
213 * @return Updated output iterator after writing the formatted string
214 */
215template <typename It>
216 requires std::output_iterator<It, char>
217inline It ToString(It out, ExitTrigger triggers, bool with_prefix = false) {
218 const std::string_view kNoneStr = with_prefix ? "ExitTrigger::None" : "None";
219
220 if (triggers == ExitTrigger::kNone) {
221 return std::format_to(out, "{}", kNoneStr);
222 }
223
224 bool first = true;
225 const auto append = [triggers, &out, &first, with_prefix](
226 std::string_view name, ExitTrigger flag) {
227 if (!HasFlag(triggers, flag)) {
228 return;
229 }
230 if (!first) {
231 out = std::format_to(out, " | ");
232 }
233 first = false;
234 out = with_prefix ? std::format_to(out, "ExitTrigger::{}", name)
235 : std::format_to(out, "{}", name);
236 };
237
238 append("PrimaryClosed", ExitTrigger::kPrimaryClosed);
239 append("AllWindowsClosed", ExitTrigger::kAllWindowsClosed);
240 append("AnyWindowClosed", ExitTrigger::kAnyWindowClosed);
241 return out;
242}
