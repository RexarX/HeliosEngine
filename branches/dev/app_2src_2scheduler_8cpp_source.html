56void Scheduler::Clear() {
57 HELIOS_ASSERT(async_loops_running_.load(std::memory_order_acquire) == 0,
58 "Cannot clear scheduler while async update loops are running! "
59 "Call Stop() or Shutdown() first.");
60 HELIOS_ASSERT(async_loop_futures_.empty(),
61 "Cannot clear scheduler while async loop futures remain! Call "
62 "Stop() or Shutdown() first.");
63 HELIOS_ASSERT(overlapping_update_futures_.empty(),
64 "Cannot clear scheduler while overlapping updates remain! Call "
65 "Stop() or Shutdown() first.");
66
67 [[maybe_unused]] const bool any_sub_app_updating =
68 std::ranges::any_of(sub_app_states_, [](const SubAppFrameState& state) {
69 return state.sub_app.get().IsUpdating();
70 });
71 HELIOS_ASSERT(!any_sub_app_updating,
72 "Cannot clear scheduler while sub-apps are updating! Call "
73 "Stop() or Shutdown() first.");
74
75 HELIOS_ASSERT(!blocking_update_future_.has_value(),
76 "Cannot clear scheduler while blocking sub-app updates are in "
77 "flight! Call Stop() or Shutdown() first.");
78
79 startup_graph_.Clear();
80 blocking_update_graph_.Clear();
81 shutdown_graph_.Clear();
82 sub_app_states_.clear();
83 blocking_update_future_.reset();
84}

92void Scheduler::Build(App& app) {
93 HELIOS_APP_PROFILE_SCOPE_N("helios::app::Scheduler::Build");
94
95 auto& main = app.GetMainSubApp();
96 auto& executor = app.GetExecutor();
97
98 main.BuildScheduler(executor);
99
100 startup_graph_.Clear();
101 blocking_update_graph_.Clear();
102 shutdown_graph_.Clear();
103 sub_app_states_.clear();
104
105 if (app.sub_apps_.empty()) {
106 return;
107 }
108
109 sub_app_states_.reserve(app.sub_apps_.size());
110
111 for (auto&& [index, sub_app] : app.sub_apps_) {
112 sub_app.SetOwnerApp(app);
113 sub_app_states_.push_back(SubAppFrameState{
114 .sub_app = sub_app,
115 .mode = ClassifySubApp(sub_app),
116 });
117
118 startup_graph_.EmplaceTask(
119 [&sub_app, &executor]() { sub_app.RunStartupUnchecked(executor); });
120
121 shutdown_graph_.EmplaceTask([&sub_app, &executor]() {
122 sub_app.WaitUntilFullyIdle();
123 sub_app.RunShutdownUnchecked(executor);
124 });
125 }
126
127 for (auto&& [index, sub_app] : app.sub_apps_) {
128 if (ClassifySubApp(sub_app) != SubAppMode::kBlocking) {
129 continue;
130 }
131
132 blocking_update_graph_.EmplaceTask(
133 [&sub_app, &executor]() { sub_app.RunUpdatePass(executor); });
134 }
135}

158void Scheduler::RunFrameOrder(App& app, const FrameOrder& order) {
159 HELIOS_APP_PROFILE_SCOPE_N("helios::app::Scheduler::RunFrameOrder");
160
161 auto& executor = app.GetExecutor();
162 auto& main = app.GetMainSubApp();
163 main.BuildScheduler(executor);
164
165 auto& world = main.GetWorld();
166 auto& ecs_scheduler = main.GetScheduler();
167
168 for (SubAppFrameState& state : sub_app_states_) {
169 state.fresh_extract_this_frame = false;
170 }
171
172 bool ran_extract = false;
173 const auto extract_type_index = ecs::StageTypeIndex::From(kExtractStage);
174
175 const auto labels = order.Labels();
176 std::optional<ecs::StageTypeIndex> last_present_stage;
177 for (auto it = labels.rbegin(); it != labels.rend(); ++it) {
178 if (ecs_scheduler.HasStage(*it)) {
179 last_present_stage = *it;
180 break;
181 }
182 }
183
184 for (const ecs::StageTypeIndex stage : labels) {
185 if (!ecs_scheduler.HasStage(stage)) {
186 continue;
187 }
188
189 ecs_scheduler.RunStage(stage, world);
190
191 const auto& stage_settings = ecs_scheduler.GetStageSettings(stage);
192 if (stage_settings.apply_commands || stage_settings.merge_messages) {
193 ecs_scheduler.ApplyStageDeferred(stage, world,
194 stage_settings.apply_commands,
195 stage_settings.merge_messages);
196 }
197 // Shared StageSettings apply to every frame order; only the last present
198 // stage in *this* order may advance message buffers (MainFrameOrder ->
199 // Extract, FramePumpOrder -> Update).
200 if (last_present_stage.has_value() && stage == *last_present_stage &&
201 stage_settings.advance_messages) {
202 world.Messages().Update();
203 }
204
205 if (stage == extract_type_index) {
206 ran_extract = true;
207 const auto& main_world = main.GetWorld();
208 for (SubAppFrameState& state : sub_app_states_) {
209 ExtractSubApp(state, main_world);
210 }
211 }
212 }
213
214 if (ran_extract) {
215 LaunchSubAppUpdates(app);
216 WaitForSubApps();
217 }
218}

274void Scheduler::LaunchSubAppUpdates(App& app) {
275 if (app.sub_apps_.empty()) {
276 return;
277 }
278
279 auto& executor = app.GetExecutor();
280
281 if (blocking_update_graph_.TaskCount() > 0) {
282 blocking_update_future_ = executor.Run(blocking_update_graph_);
283 }
284
285 PruneCompletedOverlappingUpdates();
286
287 for (SubAppFrameState& state : sub_app_states_) {
288 if (state.mode != SubAppMode::kOverlapping ||
289 !state.fresh_extract_this_frame) {
290 continue;
291 }
292
293 SubApp& sub_app = state.sub_app.get();
294 overlapping_update_futures_.push_back(executor.Async(
295 [&sub_app, &executor]() { sub_app.RunUpdatePass(executor); }));
296 }

299void Scheduler::StartAsyncUpdateLoops(App& app) {
300 auto& executor = app.GetExecutor();
301
302 for (auto&& [index, sub_app] : app.sub_apps_) {
303 if (!sub_app.IsAsync()) {
304 continue;
305 }
306
307 sub_app.ResetAsyncLoopStop();
308 async_loops_running_.fetch_add(1, std::memory_order_acq_rel);
309
310 async_loop_futures_.push_back(executor.Async([this, &sub_app, &executor]() {
311 HELIOS_DEFER {
312 async_loops_running_.fetch_sub(1, std::memory_order_acq_rel);
313 };
314
315 sub_app.RunUpdatePass(executor);
316 }));
317 }

359void Scheduler::ExtractSubApp(SubAppFrameState& state,
360 const ecs::World& main_world) {
361 SubApp& sub_app = state.sub_app.get();
362 state.fresh_extract_this_frame = false;
363
364 switch (state.mode) {
365 using enum SubAppMode;
366 case kBlocking:
367 sub_app.WaitUntilFullyIdle();
368 sub_app.Extract(main_world);
369 state.fresh_extract_this_frame = true;
370 state.consecutive_extract_skips = 0;
371 return;
372 case kAsync:
373 sub_app.Extract(main_world, true);
374 return;
375 case kOverlapping:
376 break;
377 }
378
379 if (sub_app.IsUpdating()) {
380 const size_t max_skips = sub_app.MaxExtractionSkips();
381 if (max_skips == 0 || state.consecutive_extract_skips < max_skips) {
382 ++state.consecutive_extract_skips;
383 return;
384 }
385
386 sub_app.WaitUntilFullyIdle();
387 }
388
389 sub_app.Extract(main_world);
390 state.fresh_extract_this_frame = true;
391 state.consecutive_extract_skips = 0;
392}

394auto Scheduler::ClassifySubApp(const SubApp& sub_app) noexcept
395 -> Scheduler::SubAppMode {
396 if (sub_app.IsAsync()) {
397 return SubAppMode::kAsync;
398 }
399
400 if (sub_app.AllowsOverlappingUpdates()) {
401 return SubAppMode::kOverlapping;
402 }
403
404 return SubAppMode::kBlocking;
405}
