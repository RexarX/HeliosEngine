28/// @brief MurmurHash3 finalizer; avalanches all bits of a 64-bit word.
29[[nodiscard]] constexpr uint64_t MurmurScramble64(uint64_t num) noexcept {
30 num ^= num >> 33;
31 num *= 0xff51afd7ed558ccdULL;
32 num ^= num >> 33;
33 num *= 0xc4ceb9fe1a85ec53ULL;
34 num ^= num >> 33;
35 return num;
36}

38/// @brief Reads 64 bits from `std::random_device`, combining calls if needed.
39[[nodiscard]] inline uint64_t ReadRandomDevice64() {
40 std::random_device rd{};
41 using Result = std::random_device::result_type;
42 constexpr auto result_bits = sizeof(Result) * 8U;
43 constexpr auto target_bits = sizeof(uint64_t) * 8U;
44
45 if constexpr (result_bits >= target_bits) {
46 return static_cast<uint64_t>(rd());
47 } else {
48 uint64_t value = 0;
49 auto shift = 0U;
50 while (shift < target_bits) {
51 value |= static_cast<uint64_t>(rd()) << shift;
52 shift += result_bits;
53 }
54 return value;
55 }
56}

58/**
59 * @brief Process-wide entropy captured once on first use.
60 * @details Function-local static (not a namespace-scope global) so
61 * initialization is thread-safe and avoids the static initialization order
62 * fiasco. Mixes `std::random_device` with a scrambled high-resolution clock
63 * because some platforms provide a weak `random_device`.
64 */
65[[nodiscard]] inline uint64_t CachedProcessEntropy() {
66 static const uint64_t entropy = [] {
67 uint64_t value = ReadRandomDevice64();
68 using Clock = std::chrono::high_resolution_clock;
69 const auto ticks =
70 static_cast<uint64_t>(Clock::now().time_since_epoch().count());
71 value ^= MurmurScramble64(ticks);
72 return value == 0 ? 0x9e3779b97f4a7c15ULL : value;
73 }();
74 return entropy;
75}

77/// @brief Monotonic counter so recycled OS thread ids still get unique seeds.
78[[nodiscard]] inline uint64_t NextSeedSequence() noexcept {
79 static std::atomic<uint64_t> sequence{1};
80 return sequence.fetch_add(1, std::memory_order_relaxed);
81}

88/**
89 * @brief Concept for random number engines compatible with std distributions.
90 * @details Requires presence of `result_type`, `operator()`, and
91 * `min()`/`max()` members.
92 */
93template <typename T>
94concept RandomEngine = requires(T&& engine) {
95 typename std::remove_cvref_t<T>::result_type;
96 { engine() } -> std::convertible_to<typename T::result_type>;
97 {
98 std::remove_cvref_t<T>::min()
99 } -> std::convertible_to<typename std::remove_cvref_t<T>::result_type>;
100 {
101 std::remove_cvref_t<T>::max()
102 } -> std::convertible_to<typename std::remove_cvref_t<T>::result_type>;
103};
104
105/**
106 * @brief Concept for standard-like distributions.
107 * @details Requires presence of `result_type` and callable operator(engine).
108 */
109template <typename T, typename Engine>
110concept Distribution = requires(T&& dist, Engine engine) {
111 typename std::remove_cvref_t<T>::result_type;
112 {
113 dist(engine)
114 } -> std::convertible_to<typename std::remove_cvref_t<T>::result_type>;
115};

176/**
177 * @brief Returns the process-wide 64-bit entropy word.
178 * @details Captured once on first call from `std::random_device` mixed with
179 * a high-resolution clock. Subsequent calls are free and return the same
180 * value for the lifetime of the process. Prefer `MixThreadSeed()` when
181 * constructing engines so threads do not share a stream.
182 * @return Stable per-process 64-bit seed
183 */
184[[nodiscard]] inline uint64_t RandomDeviceSeed() {
185 return details::CachedProcessEntropy();
186}
187
188/**
189 * @brief Mixes process entropy with the calling thread and a unique sequence.
190 * @details Combines `RandomDeviceSeed()`, `std::this_thread::get_id()`, and
191 * an atomic counter. The counter matters because OS thread ids can be reused
192 * after a thread exits. Each call returns a different 64-bit word, suitable
193 * as a `seed(word)` argument for xoshiro/xoroshiro (they expand it with
194 * SplitMix64 into the full state).
195 * @return Unique 64-bit seed for this call
196 */
197[[nodiscard]] inline uint64_t MixThreadSeed() {
198 const uint64_t thread_hash = static_cast<uint64_t>(
199 std::hash<std::thread::id>{}(std::this_thread::get_id()));
200 uint64_t state = details::CachedProcessEntropy();
201 state ^= details::MurmurScramble64(thread_hash);
202 state += details::NextSeedSequence() * 0x9e3779b97f4a7c15ULL;
203 const uint64_t mixed = details::MurmurScramble64(state);
204 return mixed == 0 ? thread_hash | 1U : mixed;
205}

207/**
208 * @brief Creates an engine seeded from process entropy, thread id, and
209 * sequence.
210 * @tparam Engine RandomEngine type to construct (defaults to
211 * `DefaultRandomEngine`)
212 * @return Engine instance with a unique mixed seed
213 */
214template <RandomEngine Engine = DefaultRandomEngine>
215[[nodiscard]] Engine MakeEngine() {
216 using Word = typename Engine::result_type;
217 const uint64_t seed = MixThreadSeed();
218 if constexpr (sizeof(Word) >= sizeof(uint64_t)) {
219 return Engine{static_cast<Word>(seed)};
220 } else {
221 constexpr auto word_bits = sizeof(Word) * 8U;
222 return Engine{static_cast<Word>(seed ^ (seed >> word_bits))};
223 }
224}

359template <RandomEngine Engine>
360template <utils::ArithmeticTrait T>
361inline T RandomGenerator<Engine>::Value() {
362 Engine& engine = engine_.get();
363 if constexpr (std::integral<T>) {
364 if constexpr (std::same_as<T, bool>) {
365 std::uniform_int_distribution<int> dist(0, 1);
366 return dist(engine) == 1;
367 } else {
368 std::uniform_int_distribution<T> dist(std::numeric_limits<T>::min(),
369 std::numeric_limits<T>::max());
370 return dist(engine);
371 }
372 } else {
373 std::uniform_real_distribution<T> dist(static_cast<T>(0),
374 static_cast<T>(1));
375 return dist(engine);
376 }
377}

379template <RandomEngine Engine>
380template <utils::ArithmeticTrait T, utils::ArithmeticTrait U>
381inline auto RandomGenerator<Engine>::ValueFromRange(T min, U max)
382 -> std::common_type_t<T, U> {
383 using Common = std::common_type_t<T, U>;
384 const auto cmin = static_cast<Common>(min);
385 const auto cmax = static_cast<Common>(max);
386
387 Engine& engine = engine_.get();
388 if constexpr (std::integral<Common>) {
389 using DistType =
390 std::conditional_t<std::signed_integral<Common>, int64_t, uint64_t>;
391 std::uniform_int_distribution<DistType> dist(static_cast<DistType>(cmin),
392 static_cast<DistType>(cmax));
393 return static_cast<Common>(dist(engine));
394 } else {
395 std::uniform_real_distribution<Common> dist(cmin, cmax);
396 return dist(engine);
397 }
398}
