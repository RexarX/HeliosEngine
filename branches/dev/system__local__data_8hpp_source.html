137 /**
138 * @brief Applies selected deferred local work.
139 * @details Commands and messages share one arena — `ResetArena()` runs only
140 * when `!HasPendingWork()` after the selected steps. Call `ExecuteCommands`
141 * after `World::Flush()` so reserved entities exist.
142 * @param world World to apply against
143 * @param apply_commands Whether to execute the local command queue
144 * @param merge_messages Whether to merge local messages / consumed registries
145 */
146 void Apply(World& world, bool apply_commands, bool merge_messages) {
147 HELIOS_ECS_PROFILE_SCOPE_N("helios::ecs::SystemLocalData::Apply");
148
149 if (apply_commands) {
150 ExecuteCommands(world);
151 }
152 if (merge_messages) {
153 MergeMessages(world);
154 }
155 if (!HasPendingWork()) {
156 ResetArena();
157 }
158 }

160 /**
161 * @brief Updates the system local data by executing commands and merging
162 * messages.
163 * @details Equivalent to `Apply(world, true, true)`. Call after
164 * `World::Flush()` so reserved entities exist before command execution.
165 * @param world World to update
166 */
167 void Update(World& world) {

170 }

178 /**
179 * @brief Checks whether commands or messages are still pending application.
180 * @return True if local commands, messages, or consumed-message bookkeeping
181 * remain unapplied

183 [[nodiscard]] bool HasPendingWork() const noexcept {

188 /**
189 * @brief Executes all commands in the local command queue.
190 * @param world World to execute commands on
191 */
192 void ExecuteCommands(World& world) { cmd_queue.ExecuteAll(world); }

194 /**
195 * @brief Merges messages from the local message queue into the world message
196 * manager.
197 * @details Applies consumed-message removal, then merges this system's local
198 * writes into the world current queue. Does **not** advance the
199 * previous/current lifecycle — that happens via stage
200 * `StageSettings::advance_messages` (`MessageManager::Update()`).
201 * @param world World to merge messages into
202 */
203 void MergeMessages(World& world) {
204 auto& message_manager = world.Messages();
205
206 if (!consumed_messages.Empty()) {
207 message_manager.ApplyConsumed(consumed_messages);
208 }
209 consumed_messages.Clear();
210
211 message_manager.MergeLocalMessages(std::move(message_queue));
212 message_queue.ClearAll();
213 }
214
215 /// @brief Resets the arena allocator and clears all local data.
216 /// @details Polymorphic allocator types are move-constructible but not
217 /// move-assignable, so fresh objects are constructed in place after
218 /// destroying the stale ones whose internal storage was invalidated by
219 /// the arena reset.
220 void ResetArena() noexcept {
221 std::destroy_at(&cmd_queue);
222 std::destroy_at(&message_queue);
223 std::destroy_at(&consumed_messages);
224
225 allocator.Reset();
226
227 std::construct_at(&cmd_queue, &allocator);
228 std::construct_at(&message_queue, &allocator);
229 std::construct_at(&consumed_messages, &allocator);
230 }
231
232 /// @brief Inserts or replaces the system-local arena resource.
233 void AddLocalArena() noexcept {
234 resource_manager.Insert(LocalArena{allocator});
235 }

237private:
238 /// @brief Rebuilds moved-from PMR members against the empty source allocator.
239 /// @details Must run before the destination destructor frees stolen arena
240 /// blocks. MSVC iterator debugging stores proxy nodes in the arena; leaving
241 /// them in the source queues UAF when the destination is destroyed first.
242 void ReleaseMovedFrom() noexcept {
243 Reconstruct(cmd_queue, &allocator);
244 Reconstruct(message_queue, &allocator);
245 Reconstruct(consumed_messages, &allocator);
246 resource_manager.Clear();
247 }

249 template <typename T, typename... Args>
250 static void Reconstruct(T& obj, Args&&... args) {
251 std::destroy_at(&obj);
252 std::construct_at(&obj, std::forward<Args>(args)...);
253 }
