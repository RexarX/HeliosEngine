516input::Modifiers ModifiersFromGlfw(int mods) noexcept {
517 auto result = input::Modifiers::kNone;
518 if ((mods & GLFW_MOD_SHIFT) != 0) {
519 result = result | input::Modifiers::kShift;
520 }
521 if ((mods & GLFW_MOD_CONTROL) != 0) {
522 result = result | input::Modifiers::kControl;
523 }
524 if ((mods & GLFW_MOD_ALT) != 0) {
525 result = result | input::Modifiers::kAlt;
526 }
527 if ((mods & GLFW_MOD_SUPER) != 0) {
528 result = result | input::Modifiers::kSuper;
529 }
530 if ((mods & GLFW_MOD_CAPS_LOCK) != 0) {
531 result = result | input::Modifiers::kCapsLock;
532 }
533 if ((mods & GLFW_MOD_NUM_LOCK) != 0) {
534 result = result | input::Modifiers::kNumLock;
535 }
536 return result;
537}
