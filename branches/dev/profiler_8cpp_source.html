91void Profiler::Finalize() noexcept {
92 HELIOS_ASSERT(!finalized_, "Profiler already finalized!");
93
94 size_t offset = 0;
95 const auto align_up = [](size_t value, size_t alignment) noexcept {
96 return (value + alignment - 1) & ~(alignment - 1);
97 };
98 for (auto&& [_, entry] : backends_) {
99 offset = align_up(offset, alignof(std::max_align_t));
100 entry.storage_offset = offset;
101 entry.storage_size = entry.backend->ZoneStorageSize();
102 HELIOS_ASSERT(
103 entry.storage_size <= kZoneStorageBytes,
104 "Backend '{}' ZoneStorageSize ({}) exceeds kZoneStorageBytes ({})!",
105 entry.backend->Name(), entry.storage_size, kZoneStorageBytes);
106 offset += entry.storage_size;
107 }
108
109 HELIOS_ASSERT(
110 offset <= ScopedZone::kStorage,
111 "Combined backend storage {} bytes exceeds ScopedZone::kStorage {} bytes "
112 "— raise HELIOS_PROFILE_ZONE_STORAGE_BYTES!",
113 offset, ScopedZone::kStorage);
114
115 finalized_ = true;
116 g_profiler_finalized.store(true, std::memory_order_release);
117 RegisterShutdownMemoryDispatchGuard();
118}
