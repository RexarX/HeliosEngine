25SystemLocalData::SystemLocalData(SystemLocalData&& other) noexcept
26 : allocator(ArenaCapacityForMove(other.allocator)),
27 cmd_queue(&allocator),
28 message_queue(&allocator),
29 consumed_messages(&allocator) {
30 // Do not steal `other.allocator`: PMR queues bind the *object* address as
31 // their resource. Moving the arena leaves those vectors with inverted
32 // `_M_start`/`_M_finish` (UBSan: `vector::size` / reverse iterators).
33 cmd_queue.Merge(std::move(other.cmd_queue));
34 message_queue.Merge(std::move(other.message_queue));
35 consumed_messages.MergeFrom(std::move(other.consumed_messages));
36 resource_manager = std::move(other.resource_manager);
37 AddLocalArena();
38 other.ReleaseMovedFrom();
39}

76void SystemLocalData::MergeMessages(World& world) {
