111/**
112 * @brief Trait to identify valid logger types.
113 * @details A logger type must be an object, be empty and provide:
114 * - `static constexpr std::string_view kName` variable
115 * @tparam T Type to check
116 *
117 * @code
118 * struct MyLogger {
119 * static constexpr std::string_view kName = return "MyLogger";
120 * };
121 *
122 * static_assert(LoggerTrait<MyLogger>);
123 * @endcode
124 */
125template <typename T>
126concept LoggerTrait = std::is_object_v<std::remove_cvref_t<T>> &&
127 std::is_empty_v<std::remove_cvref_t<T>> && requires {
128 {
129 std::remove_cvref_t<T>::kName
130 } -> std::convertible_to<std::string_view>;
131 };
132
133/**
134 * @brief Trait to identify loggers with custom configuration.
135 * @details A logger with config trait must satisfy `LoggerTrait` and provide:
136 * - `static Config kConfig` variable
137 * @tparam T Type to check
138 *
139 * @code
140 * struct MyLogger {
141 * static constexpr std::string_view kName = "MyLogger";
142 * static constexpr auto kConfig = Config::Default();
143 * };
144 *
145 * static_assert(LoggerWithConfigTrait<MyLogger>);
146 * @endcode
147 */
148template <typename T>
149concept LoggerWithConfigTrait = LoggerTrait<T> && requires {
150 { std::remove_cvref_t<T>::kConfig } -> std::convertible_to<Config>;
151};
152
153/**
154 * @brief Gets the name of a logger type.
155 * @details Returns the name provided by `kName` variable.
156 * @tparam T Logger type
157 * @param logger logger instance
158 * @return Name of the logger
159 *
160 * @code
161 * struct MyLogger {
162 * static constexpr std::string_view kName = "MyLogger";
163 * };
164 *
165 * constexpr auto name = LoggerNameOf<MyLogger>(); // "MyLogger"
166 * @endcode
167 */
168template <LoggerTrait T>
169[[nodiscard]] constexpr std::string_view LoggerNameOf(
170 T /*logger*/ = {}) noexcept {
171 return std::remove_cvref_t<T>::kName;
172}
173
174/**
175 * @brief Gets the configuration for a logger type.
176 * @details Returns the config provided by `kConfig` variable if available,
177 * otherwise returns the default configuration.
178 * @tparam T Logger type
179 * @param logger logger instance
180 * @return `Config` for the logger
181 *
182 * @code
183 * struct MyLogger {
184 * static constexpr std::string_view kName = "MyLogger";
185 * static constexpr auto kConfig = Config::ConsoleOnly();
186 * };
187 *
188 * constexpr auto config = LoggerConfigOf<MyLogger>();
189 * @endcode
190 */
191template <LoggerTrait T>
192[[nodiscard]] constexpr Config LoggerConfigOf(T /*logger*/ = {}) noexcept {
193 if constexpr (LoggerWithConfigTrait<T>) {
194 return std::remove_cvref_t<T>::kConfig;
195 } else {
196 return Config::Default();
197 }
198}
