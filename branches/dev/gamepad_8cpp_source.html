26void CaptureRestIfFirstSample(GamepadAxisFilter& filter, GamepadAxis axis,
27 float raw, bool auto_calibrate) noexcept {
28 const size_t index = AxisIndex(axis);
29 if (filter.seen[index] != 0) {
30 return;
31 }
32
33 filter.seen[index] = 1;
34 if (!auto_calibrate) {
35 return;
36 }
37
38 const float expected = Trigger(axis) ? GamepadAxisFilter::kTriggerRest : 0.0F;
39 if (std::fabs(raw - expected) <= GamepadAxisFilter::kRestCapture) {
40 filter.center[index] = raw;
41 }
42}

44void RecenterAxis(GamepadAxisFilter& filter, GamepadAxis axis, bool resting,
45 uint16_t rest_frames, bool auto_calibrate) noexcept {
46 const size_t index = AxisIndex(axis);
47 if (!auto_calibrate) {
48 filter.rest_frames[index] = 0;
49 return;
50 }
51
52 if (!resting) {
53 filter.rest_frames[index] = 0;
54 return;
55 }
56
57 if (filter.rest_frames[index] < rest_frames) {
58 ++filter.rest_frames[index];
59 }
60 if (filter.rest_frames[index] >= rest_frames) {
61 filter.center[index] = filter.raw.Get(axis);
62 filter.rest_frames[index] = 0;
63 }
64}

66void RecenterStick(GamepadAxisFilter& filter, GamepadAxis x_axis,
67 GamepadAxis y_axis, const AxisFilter& stick,
68 uint16_t rest_frames, bool auto_calibrate) noexcept {
69 const float dx = filter.raw.Get(x_axis) - filter.center[AxisIndex(x_axis)];
70 const float dy = filter.raw.Get(y_axis) - filter.center[AxisIndex(y_axis)];
71 const bool resting = std::hypot(dx, dy) < stick.deadzone;
72 RecenterAxis(filter, x_axis, resting, rest_frames, auto_calibrate);
73 RecenterAxis(filter, y_axis, resting, rest_frames, auto_calibrate);
74}

210 }
