26[[nodiscard]] bool RegisterFramePaceSchedule(ecs::Scheduler& scheduler,
27 bool in_update_stage) {
28 if (scheduler.TryGetSchedule(kFramePace) != nullptr) {
29 return false;
30 }
31
32 auto schedule = ecs::Schedule::From(kFramePace);
33 schedule.Settings().executor_kind = ecs::ExecutorKind::kMainThread;
34
35 if (in_update_stage) {
36 scheduler.Add(kFramePace, std::move(schedule))
37 .InStage(kUpdateStage)
38 .Before(kFirst);
39 return true;
40 }
41
42 if (!scheduler.HasStage(kFramePaceStage)) {
43 scheduler.AddStage(kFramePaceStage);
44 }
45 scheduler.OrderStage(kFramePaceStage).Before(kUpdateStage);
46 scheduler.Add(kFramePace, std::move(schedule)).InStage(kFramePaceStage);
47 return true;
48}

52auto FrameLimiter::ResolveInterval() const noexcept
53 -> std::optional<std::chrono::nanoseconds> {
54 const auto mode = Mode();
55 if (mode == FrameLimiterMode::kOff) {
56 return std::nullopt;
57 }
58
59 if (mode == FrameLimiterMode::kAuto) {
60 const uint32_t hz = RefreshRate();
61 if (hz == 0) {
62 return std::nullopt;
63 }
64 constexpr int64_t kNanosecsInSec = 1'000'000'000;
65 return std::chrono::nanoseconds{kNanosecsInSec / hz};
66 }
67
68 const auto interval = Interval();
69 if (interval.count() <= 0) {
70 return std::nullopt;
71 }
72 return interval;
73}

75void FrameLimiter::Wait() {
76 HELIOS_APP_PROFILE_SCOPE_N("helios::app::FrameLimiter::Wait");
77
78 const auto interval = ResolveInterval();
79 if (!interval.has_value()) {
80 next_tick_.reset();
81 return;
82 }
83
84 const auto now = Clock::now();
85 if (!next_tick_.has_value()) {
86 next_tick_ = now + *interval;
87 return;
88 }
89
90 utils::PreciseSleepUntil(*next_tick_);
91 next_tick_ = std::max(*next_tick_, Clock::now()) + *interval;
92}
