10void TemporaryStorage::Init() {
11 // The node is heap-allocated and intentionally never deleted.
12 registry_node_ = new RegistryNode{};
13 registry_node_->owner = this;
14
15 // Treiber-stack push of registry_node_ onto registry_head_.
16 // registry_node_->next is written here and never again, so once this
17 // node is visible to another thread (via the successful CAS's release,
18 // paired with an acquire load of registry_head_ in ResetAll), `next` is
19 // guaranteed visible too.
20 registry_node_->next = registry_head_.load(std::memory_order_relaxed);
21 while (!registry_head_.compare_exchange_weak(
22 registry_node_->next, registry_node_, std::memory_order_release,
23 std::memory_order_relaxed)) {
24 // On failure, compare_exchange_weak has already reloaded the current
25 // head into registry_node_->next, so we just retry the CAS with it.
26 }

29void TemporaryStorage::Destroy() noexcept {
30 // Claim exclusive access to this node before letting member destruction
31 // (upstream_resource_'s teardown, which runs immediately after this
32 // function body returns) proceed.
33 auto expected = NodeState::kActive;
34 while (!registry_node_->state.compare_exchange_weak(
35 expected, NodeState::kBusy, std::memory_order_acq_rel,
36 std::memory_order_relaxed)) {
37 expected = NodeState::kActive;
38 std::this_thread::yield();
39 }
40
41 // We now hold kBusy exclusively. upstream_resource_'s destructor runs
42 // implicitly right after this function body, tearing down the chunk
43 // list with no other thread able to touch it concurrently.
44 registry_node_->state.store(NodeState::kRetired, std::memory_order_release);
45}

47void TemporaryStorage::ResetAllImpl() noexcept {
48 // Acquire pairs with the release in the constructor's successful CAS:
49 // if we observe a node here, we also observe its fully-initialized
50 // `next` and `owner`.
51 for (RegistryNode* node = registry_head_.load(std::memory_order_acquire);
52 node != nullptr; node = node->next) {
53 // Try to claim the node. Success means we have exclusive rights to
54 // call Reset() on its owner, in particular, the owning thread's
55 // destructor cannot be mid-teardown of the same upstream_resource_,
56 // because it would have had to win this same CAS first. Failure means
57 // the node is either already kBusy (most likely: its owner is
58 // concurrently exiting) or kRetired; either way we skip it rather than
59 // wait.
60 auto expected = NodeState::kActive;
61 if (node->state.compare_exchange_strong(expected, NodeState::kBusy,
62 std::memory_order_acq_rel,
63 std::memory_order_acquire)) {
64 node->owner->Reset();
65 node->state.store(NodeState::kActive, std::memory_order_release);
66 }
67 }
