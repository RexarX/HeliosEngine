102TemporaryStorage::ThreadBinding::~ThreadBinding() noexcept {
103 // Heap-free TLS teardown: CAS only. Never overwrite kBusy with kRetired
104 // while ResetAll() is inside Release(), that would let ClaimSlot reclaim
105 // the slot concurrently. kBusyRetired tells ResetAll to publish kRetired
106 // after Release() finishes.
107 Slot& slot = g_slots[slot_index];
108 for (;;) {
109 auto current = slot.state.load(std::memory_order_acquire);
110 if (current == SlotState::kActive) {
111 if (slot.state.compare_exchange_weak(current, SlotState::kRetired,
112 std::memory_order_acq_rel,
113 std::memory_order_acquire)) {
114 return;
115 }
116 std::this_thread::yield();
117 continue;
118 }
119
120 if (current == SlotState::kBusy) {
121 if (slot.state.compare_exchange_weak(current, SlotState::kBusyRetired,
122 std::memory_order_acq_rel,
123 std::memory_order_acquire)) {
124 return;
125 }
126 std::this_thread::yield();
127 continue;
128 }
129 return;
130 }
131}

138size_t TemporaryStorage::ClaimSlot() noexcept {
139 HELIOS_MEMORY_PROFILE_SCOPE_N("helios::mem::TemporaryStorage::ClaimSlot");
140
141 for (size_t i = 0; i < g_slots.size(); ++i) {
142 Slot& slot = g_slots[i];
143
144 auto expected = SlotState::kFree;
145 if (slot.state.compare_exchange_strong(expected, SlotState::kActive,
146 std::memory_order_acq_rel,
147 std::memory_order_acquire)) {
148 slot.block.Construct();
149 return i;
150 }
151
152 expected = SlotState::kRetired;
153 if (slot.state.compare_exchange_strong(expected, SlotState::kActive,
154 std::memory_order_acq_rel,
155 std::memory_order_acquire)) {
156 // Reclaiming a dead thread's slot on the living claimant: overflow
157 // blocks from the previous occupant are freed here, never in a TLS
158 // destructor.
159 slot.block.Release();
160 return i;
161 }
162 }
163
164 HELIOS_VERIFY(false,
165 "TemporaryStorage slot pool exhausted (kMaxThreads={}); "
166 "increase TemporaryStorage::kMaxThreads",
167 kMaxThreads);
168 std::abort();
169}

180void TemporaryStorage::ResetAll() noexcept {
181 HELIOS_MEMORY_PROFILE_SCOPE_N("helios::mem::TemporaryStorage::ResetAll");
182
183 for (Slot& slot : g_slots) {
184 auto expected = SlotState::kActive;
185 if (!slot.state.compare_exchange_strong(expected, SlotState::kBusy,
186 std::memory_order_acq_rel,
187 std::memory_order_acquire)) {
188 continue;
189 }
190
191 slot.block.Release();
192
193 // Restore kActive iff we still hold kBusy. Failure means the owner
194 // CAS-ed kBusy -> kBusyRetired; publish kRetired so ClaimSlot can
195 // reclaim after this Release() has finished.
196 expected = SlotState::kBusy;
197 if (!slot.state.compare_exchange_strong(expected, SlotState::kActive,
198 std::memory_order_release,
199 std::memory_order_acquire)) {
200 slot.state.store(SlotState::kRetired, std::memory_order_release);
201 }
202 }
