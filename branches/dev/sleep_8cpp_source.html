27[[nodiscard]] HANDLE HighResolutionTimer() {
28 // Intentionally never closed: thread_local storage duration means the OS
29 // reclaims this handle on thread exit. Reused across calls on the same
30 // thread to avoid per-call CreateWaitableTimerEx overhead.
31 thread_local HANDLE timer = []() -> HANDLE {
32 HANDLE handle = CreateWaitableTimerExW(
33 nullptr, nullptr, CREATE_WAITABLE_TIMER_HIGH_RESOLUTION,
34 TIMER_ALL_ACCESS);
35 if (handle == nullptr) {
36 handle = CreateWaitableTimerW(nullptr, TRUE, nullptr);
37 }
38 return handle;
39 }();
40 return timer;
41}

59void UnixSleepFor(std::chrono::nanoseconds duration) {
60 timespec remaining{
61 .tv_sec = static_cast<time_t>(duration.count() / 1'000'000'000),
62 .tv_nsec = static_cast<long>(duration.count() % 1'000'000'000),
63 };
64
65 while (true) {
66 timespec requested = remaining;
67 if (nanosleep(&requested, &remaining) == 0) {
68 return;
69 }
70 if (errno != EINTR) {
71 const auto remaining_ns =
72 std::chrono::nanoseconds{remaining.tv_sec} * 1'000'000'000 +
73 std::chrono::nanoseconds{remaining.tv_nsec};
74 std::this_thread::sleep_for(remaining_ns);
75 return;

82void LinuxSleepUntil(std::chrono::steady_clock::time_point deadline) {
83 const auto now = std::chrono::steady_clock::now();
84 if (now >= deadline) {
85 return;
86 }
87
88 const auto remaining =
89 std::chrono::duration_cast<std::chrono::nanoseconds>(deadline - now);
90
91 timespec ts{};
92 if (clock_gettime(CLOCK_MONOTONIC, &ts) != 0) {
93 UnixSleepFor(remaining);
94 return;
95 }
96
97 ts.tv_sec += static_cast<time_t>(remaining.count() / 1'000'000'000);
98 ts.tv_nsec += static_cast<long>(remaining.count() % 1'000'000'000);
99 if (ts.tv_nsec >= 1'000'000'000L) {
100 ++ts.tv_sec;
101 ts.tv_nsec -= 1'000'000'000L;
102 }
103
104 while (true) {
105 const int err =
106 clock_nanosleep(CLOCK_MONOTONIC, TIMER_ABSTIME, &ts, nullptr);
107 if (err == 0) {
108 return;
109 }
110 if (err != EINTR) {
111 UnixSleepFor(remaining);
112 return;
113 }
114 }
115}

150void PreciseSleepUntil(std::chrono::steady_clock::time_point deadline,
151 bool pinned_thread) {
152 using Clock = std::chrono::steady_clock;
153
154 auto current = Clock::now();
155 if (current >= deadline) {
156 return;
157 }
158
159 constexpr auto kShortSleep = 1'000'000ns;
160 constexpr auto kMaxSleepCap = 3'000'000ns;
161 auto max_sleep = kShortSleep;
162
163 while (current + max_sleep < deadline) {
164 SleepFor(kShortSleep);
165 const auto now = Clock::now();
166 const auto next_sleep = now - current;
167 max_sleep = std::clamp(next_sleep, kShortSleep, kMaxSleepCap);
168 current = now;
169 }
170
171 if (current < deadline) {
172 const auto remaining = deadline - current;
173 const auto overshoot = max_sleep - kShortSleep;
174 if (remaining > overshoot) {
175 SleepFor(std::chrono::duration_cast<std::chrono::nanoseconds>(remaining -
176 overshoot));
177 current = Clock::now();
178 }
179 }
180
181 while (current + kShortSleep < deadline) {
182 SleepFor(kShortSleep);
183 current = Clock::now();
184 }
185
186 while (current < deadline) {
187 if (pinned_thread) [[likely]] {
188 HELIOS_PAUSE_CPU();
189 } else {
190 std::this_thread::yield();
191 }
192 current = Clock::now();
193 }
194}
