31/**
32 * @brief Returns the index of the first id >= `last_message_count`.
33 * @param ids Sorted message ids for one buffer
34 * @param last_message_count Cursor position
35 * @return Offset into `ids`, or `ids.size()` when all ids are older
36 */
37template <MessageTrait T>
38[[nodiscard]] constexpr size_t MessageUnreadOffset(
39 std::span<const AnyMessageId> ids,
40 MessageId<T> last_message_count) noexcept {
41 const AnyMessageId cursor_pos = last_message_count.ToAny();
42 const auto it = std::ranges::lower_bound(ids, cursor_pos);
43 return static_cast<size_t>(it - ids.begin());
44}
