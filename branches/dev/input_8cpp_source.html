95void EmitGamepadState(int jid, const GLFWgamepadstate& state,
96 GamepadSlotCache& slot, input::GamepadWriters& writers,
97 bool seed) {
98 for (size_t i = 0; i < slot.buttons.size(); ++i) {
99 const unsigned char next = state.buttons[i];
100 if (!seed && next == slot.buttons[i]) {
101 continue;
102 }
103 slot.buttons[i] = next;
104 if (seed && next != GLFW_PRESS) {
105 continue;
106 }
107 writers.buttons.Write({
108 .id = jid,
109 .button = GamepadButtonFromGlfw(static_cast<int>(i)),
110 .state = next == GLFW_PRESS ? input::ButtonState::kPressed
111 : input::ButtonState::kReleased,
112 });
113 }
114
115 for (size_t i = 0; i < slot.axes.size(); ++i) {
116 const float next = state.axes[i];
117 if (!seed && next == slot.axes[i]) {
118 continue;
119 }
120 slot.axes[i] = next;
121 writers.axes.Write({
122 .id = jid,
123 .axis = GamepadAxisFromGlfw(static_cast<int>(i)),
124 .value = next,
125 });
126 }
127}
