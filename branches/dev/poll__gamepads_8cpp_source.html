140void EmitGamepadState(int slot_id, GamepadSlotCache& slot,
141 helios::input::GamepadWriters& writers, bool seed) {
142 SDL_Gamepad* gamepad = slot.gamepad;
143 if (gamepad == nullptr) {
144 return;
145 }
146
147 for (int button = 0; button < SDL_GAMEPAD_BUTTON_COUNT; ++button) {
148 const auto mapped =
149 GamepadButtonFromSdl(static_cast<SDL_GamepadButton>(button));
150 if (!mapped.has_value()) {
151 continue;
152 }
153 const size_t index = static_cast<size_t>(*mapped);
154 const bool next =
155 SDL_GetGamepadButton(gamepad, static_cast<SDL_GamepadButton>(button));
156 if (!seed && next == slot.pad_buttons[index]) {
157 continue;
158 }
159 slot.pad_buttons[index] = next;
160 if (seed && !next) {
161 continue;
162 }
163 writers.buttons.Write({
164 .id = slot_id,
165 .button = *mapped,
166 .state = next ? helios::input::ButtonState::kPressed
167 : helios::input::ButtonState::kReleased,
168 });
169 }
170
171 for (size_t i = 0; i < slot.pad_axes.size(); ++i) {
172 const auto axis = static_cast<helios::input::GamepadAxis>(i);
173 const float next = AxisValue(gamepad, axis);
174 if (!seed && next == slot.pad_axes[i]) {
175 continue;
176 }
177 slot.pad_axes[i] = next;
178 writers.axes.Write({.id = slot_id, .axis = axis, .value = next});
179 }
180
181 const helios::input::GamepadPower power = PowerFromSdl(gamepad);
182 if (seed || power.state != slot.power.state ||
183 power.percent != slot.power.percent) {
184 slot.power = power;
185 writers.power.Write({.id = slot_id, .power = power});
186 }
187
188 const int touchpads = SDL_GetNumGamepadTouchpads(gamepad);
189 if (touchpads > 0) {
190 int fingers = SDL_GetNumGamepadTouchpadFingers(gamepad, 0);
191 slot.touchpad_count =
192 ClampCount(fingers, helios::input::Gamepad::kMaxTouchpadFingers,
193 "touchpad finger");
194 for (uint8_t finger = 0; finger < slot.touchpad_count; ++finger) {
195 bool down = false;
196 float x = 0.0F;
197 float y = 0.0F;
198 float pressure = 0.0F;
199 SDL_GetGamepadTouchpadFinger(gamepad, 0, finger, &down, &x, &y,
200 &pressure);
201 auto& cached = slot.touchpad[finger];
202 if (!seed && cached.down == down && cached.x == x && cached.y == y &&
203 cached.pressure == pressure) {
204 continue;
205 }
206 cached = {.x = x, .y = y, .pressure = pressure, .down = down};
207 writers.touchpad.Write({
208 .id = slot_id,
209 .x = x,
210 .y = y,
211 .pressure = pressure,
212 .finger = finger,
213 .down = down,
214 });
215 }

370void RemapDevice(GamepadCache& cache, SDL_JoystickID instance_id,
371 helios::input::GamepadWriters& gamepads,
372 helios::input::JoystickWriters& sticks) {
373 const int slot_id = FindSlotByInstance(cache, instance_id);
374 if (slot_id < 0) {
375 OpenDevice(cache, instance_id, gamepads, sticks);
376 return;
377 }
378
379 GamepadSlotCache& slot = cache.slots[static_cast<size_t>(slot_id)];
380 const bool mapped = SDL_IsGamepad(instance_id);
381 if (mapped && !slot.is_gamepad) {
382 RemoveDevice(cache, instance_id, gamepads, sticks);
383 OpenDevice(cache, instance_id, gamepads, sticks);
384 return;
385 }
386 if (!mapped && slot.is_gamepad) {
387 RemoveDevice(cache, instance_id, gamepads, sticks);
388 OpenDevice(cache, instance_id, gamepads, sticks);
389 return;
390 }
391 if (mapped && slot.gamepad != nullptr) {
392 const char* mapping = SDL_GetGamepadMapping(slot.gamepad);
393 gamepads.remapped.Write({
394 .id = slot_id,
395 .mapping = mapping != nullptr ? mapping : "",
396 });
397 EmitGamepadState(slot_id, slot, gamepads, true);
398 }

444 helios::input::GamepadWriters gamepads,

446 ecs::Res<GamepadCache> cache) const {

448 return;
449 }
450
451 if (!cache->enumerated) {
452 EnumerateDevices(*cache, gamepads, sticks);
453 cache->enumerated = true;
454 }
455

461

474 for (size_t slot_index = 0; slot_index < cache->slots.size(); ++slot_index) {
475 GamepadSlotCache& slot = cache->slots[slot_index];
476 if (!slot.connected) {
477 continue;

480 if (slot.is_gamepad) {
