114/**
115 * @brief Applies a 1D deadzone and optional rescale.
116 * @details Values with magnitude below `deadzone` become `0`. Magnitudes at or
117 * above `livezone` saturate to `+/- 1` when `rescale` is set, otherwise to
118 * `+/- livezone`. Between the two, `rescale` maps the remaining range onto
119 * `[0, 1]` (preserving sign).
120 * @param value Axis value, typically in `[-1, 1]` or `[0, 1]`
121 * @param deadzone Inner cutoff (exclusive)
122 * @param livezone Outer saturate magnitude
123 * @param rescale Remap `(deadzone, livezone)` onto `(0, 1)`
124 * @return Filtered value
125 */
126[[nodiscard]] constexpr float ApplyLinearDeadzone(float value, float deadzone,
127 float livezone,
128 bool rescale) noexcept {
129 if (livezone <= deadzone) {
130 return 0.0F;
131 }
132
133 const float mag = std::fabs(value);
134 if (mag < deadzone) {
135 return 0.0F;
136 }
137
138 const float sign = (value < 0.0F) ? -1.0F : 1.0F;
139 if (mag >= livezone) {
140 return rescale ? sign : sign * livezone;
141 }
142
143 if (!rescale) {
144 return value;
145 }
146 return sign * (mag - deadzone) / (livezone - deadzone);
147}

149/**
150 * @brief Applies `ApplyLinearDeadzone` using an `AxisFilter`.
151 * @param value Axis value
152 * @param filter Deadzone settings
153 * @return Filtered value
154 */
155[[nodiscard]] constexpr float ApplyLinearDeadzone(
156 float value, const AxisFilter& filter) noexcept {
157 return ApplyLinearDeadzone(value, filter.deadzone, filter.livezone,

161/**
162 * @brief Applies a circular (radial) deadzone to a stick pair.
163 * @details If the vector length is below `deadzone`, returns `(0, 0)`. At or
164 * above `livezone`, `rescale` saturates onto the unit circle; otherwise the
165 * vector is scaled to length `livezone`. Between the two, `rescale` maps
166 * length onto `(0, 1]` along the same direction.
167 * @note When `livezone <= deadzone`, the whole range is treated as dead.
168 * @param x X component
169 * @param y Y component
170 * @param deadzone Inner radial cutoff
171 * @param livezone Outer radial saturate length
172 * @param rescale Remap `(deadzone, livezone)` onto `(0, 1]`
173 * @return Filtered `(x, y)` pair
174 */
175[[nodiscard]] constexpr auto ApplyRadialDeadzone(float x, float y,
176 float deadzone, float livezone,
177 bool rescale)
178 -> std::pair<float, float> {
179 if (livezone <= deadzone) {
180 return {0.0F, 0.0F};
181 }
182
183 const float len = std::hypot(x, y);
184 if (len < deadzone || len == 0.0F) {
185 return {0.0F, 0.0F};
186 }
187
188 float scale = 1.0F;
189 if (len >= livezone) {
190 scale = rescale ? (1.0F / len) : (livezone / len);
191 } else if (rescale) {
192 scale = (len - deadzone) / (livezone - deadzone) / len;
193 }
194
195 return {x * scale, y * scale};
196}

199 * @brief Applies `ApplyRadialDeadzone` using an `AxisFilter`.

201 * @param y Y component
202 * @param filter Deadzone settings
203 * @return Filtered `(x, y)` pair
204 */
205[[nodiscard]] constexpr auto ApplyRadialDeadzone(float x, float y,
206 const AxisFilter& filter)
207 -> std::pair<float, float> {

212/**
213 * @brief Remaps a GLFW-style trigger axis to `[0, 1]`.
214 * @details GLFW rest is `-1` and fully pressed is `+1`. `center` is the
215 * calibrated rest value: `t = (raw - center) / (1 - center)`.
216 * @param raw Raw trigger value
217 * @param center Rest-center (typically `-1`)
218 * @return Value clamped to `[0, 1]`, or `0` when `center >= 1`
219 */
220[[nodiscard]] constexpr float RemapTrigger(float raw, float center) noexcept {
221 const float denom = 1.0F - center;
222 if (denom <= 0.0F) {
223 return 0.0F;
224 }
225 return std::clamp((raw - center) / denom, 0.0F, 1.0F);
226}
