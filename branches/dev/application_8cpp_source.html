168auto App::ShouldExit() const noexcept -> std::optional<ExitCode> {
169 const auto& world = GetWorld();
170 if (!world.HasMessage<AppExit>()) [[unlikely]] {
171 return std::nullopt;
172 }
173
174 // Fresh IncludeBacklog cursor: poll currently retained AppExit messages each
175 // check without implying once-only delivery across ShouldExit calls.
176 auto cursor = ecs::MessageCursor<AppExit>::IncludeBacklog();
177 const auto reader = world.ReadMessages<AppExit>(cursor);
178 if (reader.Empty()) {
179 return std::nullopt;
180 }
181
182 const bool any_failure = std::ranges::any_of(
183 reader, [](auto wrapper) { return wrapper->code == ExitCode::kFailure; });
184
185 return any_failure ? ExitCode::kFailure : ExitCode::kSuccess;
186}

294void App::WaitForPluginsReady() {
295 HELIOS_APP_PROFILE_SCOPE_N("helios::app::App::WaitForPluginsReady");
296
297 log::Info("Waiting for plugins to be ready...");
298
299 if (PluginsReady()) {
300 return;
301 }
302
303 auto sleep_duration = kPluginReadyPollMinSleep;
304
305 while (!PluginsReady()) {
306 PollPlugins();
307
308 {
309 std::unique_lock lock(plugins_ready_mutex_);
310 plugins_ready_cv_.wait_for(lock, sleep_duration,
311 [this] { return PluginsReady(); });
312 }
313
314 sleep_duration = std::min(sleep_duration * 2, kPluginReadyPollMaxSleep);
315 }
316}

318bool App::PluginsReady() const {
319 for (const auto& [_, storage] : plugins_) {
320 if (!storage.plugin->IsReady(*this)) {
321 return false;
322 }
323 }
324
325 return std::ranges::all_of(dynamic_plugins_, [this](const auto& pair) {
326 return !pair.second.Loaded() || pair.second.GetPlugin().IsReady(*this);
327 });
