40void Plugin::Finish(app::App& app) {
41 auto& world = app.GetWorld();
42 auto& context = world.WriteResource<Context>();
43#ifdef HELIOS_MODULE_INPUT_AVAILABLE
44 // WindowInputPlugin builds glfw before input; detect input after all Builds.
45 // HELIOS_MODULE_INPUT_AVAILABLE only means the module is linked.
46 // WindowPlugin apps must not run systems that require input::Settings.
47 context.input_enabled = world.HasMessage<input::KeyboardInputMsg>();
48 if (context.input_enabled) {
49 app.TryInsertResources(GamepadCache{}, CursorCache{});
50 app.AddSystems(window::kEvents, PollGamepads{}, ApplyCursors{},
51 ApplyRawMouseMotion{})
52 .Sequence()
53 .After<PollEvents>()
54 .Before<ApplyChanges>();
55 }
56#endif
57 context.frame_pump_user_data = &app;
58 // Size/pos/scale callbacks invoke this while glfwPollEvents is on the stack
59 // (OS modal resize/move). FramePumpOrder is Update only so it cannot re-enter
60 // kEvents. WM_PAINT/refresh must not trigger it.
61 context.frame_pump = [](void* user_data) {
62 auto& application = *static_cast<app::App*>(user_data);
63 application.RunFrameOrder(
64 application.GetWorld().ReadResource<app::FramePumpOrder>());
65 };
66}
