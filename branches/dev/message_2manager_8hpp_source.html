33/**
34 * @brief Central coordinator for message lifecycle with double buffering and
35 * consumed message removal.
36 * @details Implements message management with:
37 * - Double buffering: messages persist for one full update cycle (two frames)
38 * - Monotonic per-type message ids assigned on entry to the global queues
39 * - Explicit registration: messages must be registered before use
40 * - Automatic clearing: messages are cleared after their lifecycle expires
41 * - Manual control: users can opt-out of auto-clearing
42 * - Consumed message removal: at Update time, consumed messages are physically
43 * removed from queues without additional heap allocation
44 *
45 * Message Lifecycle:
46 * - Frame N: Messages written to current queue (ids assigned)
47 * - Frame N+1: Messages readable from previous queue (after swap)
48 * - Frame N+2: Messages cleared from previous queue (automatic policy)
49 *
50 * Delivery to systems is tracked separately via `MessageCursor<T>`. Retention
51 * does not imply re-delivery to a cursor that already observed a message.
52 *
53 * Consumed messages are removed at Update time based on merged per-system
54 * `ConsumedMessagesRegistry` instances. Each system writes to its own registry
55 * during parallel execution, and all registries are merged and applied here.
56 *
57 * @note Partially thread-safe.
58 */

458 HELIOS_ECS_PROFILE_SCOPE_N("helios::ecs::MessageManager::Update");

469inline void MessageManager::Update() {
470 HELIOS_ECS_PROFILE_SCOPE_N("helios::ecs::MessageManager::Update");
471
472 for (const auto& [type_index, metadata] : registered_messages_) {
473 if (!metadata.is_async &&
474 metadata.clear_policy == MessageClearPolicy::kAutomatic) {
475 previous_messages_.Clear(type_index);
476 if (auto* ids = previous_ids_.TryGet(type_index)) {
477 ids->clear();
478 }
479 }
480 }
481
482 previous_messages_.Merge(current_messages_);
483 AgeIds();
484 current_messages_.ClearAll();
485}

487template <typename Alloc>
488inline void MessageManager::ApplyConsumed(
489 const ConsumedMessagesRegistry<Alloc>& merged_consumed) {
490 HELIOS_ECS_PROFILE_SCOPE_N("helios::ecs::MessageManager::ApplyConsumed");
491 HELIOS_ECS_PROFILE_ZONE_VALUE(merged_consumed.TotalConsumedCount());
492
493 for (const auto& [type_index, consumed_ids] : merged_consumed.Data()) {
494 if (consumed_ids.empty()) {
495 continue;
496 }
497
498 const auto* metadata = registered_messages_.TryGet(type_index);
499 if (metadata == nullptr || metadata->is_async) {
500 continue;
501 }
502
503 auto* prev_ids = previous_ids_.TryGet(type_index);
504 auto* curr_ids = current_ids_.TryGet(type_index);
505 if ((prev_ids == nullptr || prev_ids->empty()) &&
506 (curr_ids == nullptr || curr_ids->empty())) {
507 continue;
508 }
509
510 std::vector<size_type> prev_indices;
511 std::vector<size_type> curr_indices;
512 prev_indices.reserve(consumed_ids.size());
513 curr_indices.reserve(consumed_ids.size());
514
515 for (const AnyMessageId id : consumed_ids) {
516 if (prev_ids != nullptr) {
517 const auto it = std::ranges::lower_bound(*prev_ids, id);
518 if (it != prev_ids->end() && *it == id) {
519 prev_indices.push_back(
520 static_cast<size_type>(it - prev_ids->begin()));
521 continue;
522 }
523 }
524 if (curr_ids != nullptr) {
525 const auto it = std::ranges::lower_bound(*curr_ids, id);
526 if (it != curr_ids->end() && *it == id) {
527 curr_indices.push_back(
528 static_cast<size_type>(it - curr_ids->begin()));
529 }
530 }
531 }
532
533 if (!prev_indices.empty()) {
534 previous_messages_.RemoveIndices(type_index, prev_indices);
535 RemoveIds(type_index, *prev_ids, prev_indices);
536 }
537 if (!curr_indices.empty()) {
538 current_messages_.RemoveIndices(type_index, curr_indices);
539 RemoveIds(type_index, *curr_ids, curr_indices);
540 }
541 }

544template <AnyMessageTrait... Ts>
545 requires(sizeof...(Ts) > 0)
546inline void MessageManager::Register() {
547 [[maybe_unused]] constexpr std::array type_indices = {
548 MessageTypeIndex::From<Ts>()...};
549 [[maybe_unused]] constexpr std::array names = {MessageNameOf<Ts>()...};
550
551#ifdef HELIOS_ENABLE_ASSERTS
552 std::string already_registered;
553 for (size_t i = 0; i < sizeof...(Ts); ++i) {
554 if (!registered_messages_.Contains(type_indices[i])) {
555 continue;
556 }
557
558 if (!already_registered.empty()) {
559 already_registered.append(", ");
560 }
561 already_registered.append(names[i]);
562 }
563
564 HELIOS_ASSERT(already_registered.empty(),
565 "Message type(s) '{}' already registered!", already_registered);
566#endif
567
568 (
569 [this]<typename T>() {
570 registered_messages_.template Emplace<T>(
571 MessageMetadata{.type_index = MessageTypeIndex::From<T>(),
572 .name = MessageNameOf<T>(),
573 .clear_policy = MessageClearPolicyOf<T>(),
574 .is_async = AsyncMessageTrait<T>});
575
576 if constexpr (AsyncMessageTrait<T>) {
577 async_messages_.template Register<T>();
578 } else if constexpr (MessageTrait<T>) {
579 current_messages_.template Register<T>();
580 previous_messages_.template Register<T>();
581 current_ids_.template Ensure<T>();
582 previous_ids_.template Ensure<T>();
583 message_counts_.template Ensure<T>() = 0;
584 }
585 }.template operator()<Ts>(),
586 ...);
587}

589template <MessageTrait T>
590inline auto MessageManager::Write(T&& message)
591 -> MessageId<std::remove_cvref_t<T>> {
592 using DecayedT = std::remove_cvref_t<T>;
593 HELIOS_ASSERT(
594 registered_messages_.Contains(MessageTypeIndex::From<DecayedT>()),
595 "Message type '{}' is not registered!", MessageNameOf<DecayedT>());
596
597 auto& next_id = message_counts_.template Get<DecayedT>();
598 const MessageId<DecayedT> id{next_id++};
599 current_messages_.Enqueue(std::forward<T>(message));
600 current_ids_.template Get<DecayedT>().push_back(AnyMessageId::From(id));
601 return id;
602}

717inline auto MessageManager::UnreadCount(
718 MessageId<T> last_message_count) const noexcept -> size_type {
719 const AnyMessageId cursor_pos = last_message_count.ToAny();
720 const auto count_from = [cursor_pos](std::span<const AnyMessageId> ids) {
721 const auto it = std::ranges::lower_bound(ids, cursor_pos);
722 return static_cast<size_type>(ids.end() - it);
723 };
724 return count_from(PreviousIds<T>()) + count_from(CurrentIds<T>());
725}

727inline void MessageManager::AssignIds(MessageTypeIndex type_index,
728 size_type count) {
729 if (count == 0) {
730 return;
731 }
732
733 auto& next_id = message_counts_.Ensure(type_index);
734 auto& ids = current_ids_.Ensure(type_index);
735 ids.reserve(ids.size() + count);
736 for (size_type i = 0; i < count; ++i) {
737 ids.push_back(AnyMessageId{.value = next_id++, .type = type_index});
738 }
739}

755inline void MessageManager::AgeIds() {
756 for (auto&& [type_index, curr_ids] : current_ids_) {
757 auto& prev_ids = previous_ids_.Ensure(type_index);
758 prev_ids.insert(prev_ids.end(), curr_ids.begin(), curr_ids.end());
759 curr_ids.clear();

763inline void MessageManager::RemoveIds(
764 MessageTypeIndex /*type_index*/, MessageIdList& ids,
765 std::span<const size_type> sorted_indices) {
766 if (sorted_indices.empty()) {
767 return;
768 }
769
770 size_type idx = sorted_indices.size();
771 while (idx > 0) {
772 --idx;
773 size_type range_end = sorted_indices[idx] + 1;
774 size_type range_start = sorted_indices[idx];
775 while (idx > 0 && sorted_indices[idx - 1] == range_start - 1) {
776 --idx;
777 range_start = sorted_indices[idx];
778 }
779 ids.erase(ids.begin() + static_cast<ptrdiff_t>(range_start),
780 ids.begin() + static_cast<ptrdiff_t>(range_end));
781 }
782}
