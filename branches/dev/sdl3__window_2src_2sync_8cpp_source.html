60[[nodiscard]] uint32_t RefreshRateToHz(const SDL_DisplayMode& mode) {
61 if (mode.refresh_rate > 0.0F) {
62 return static_cast<uint32_t>(mode.refresh_rate);
63 }
64 if (mode.refresh_rate_denominator != 0) {
65 return static_cast<uint32_t>(
66 static_cast<double>(mode.refresh_rate_numerator) /
67 static_cast<double>(mode.refresh_rate_denominator));
68 }

72[[nodiscard]] bool FindFullscreenMode(SDL_DisplayID display, uint32_t width,
73 uint32_t height, uint32_t refresh_rate,
74 SDL_DisplayMode& out_mode) {
75 int mode_count = 0;
76 SDL_DisplayMode** modes = SDL_GetFullscreenDisplayModes(display, &mode_count);
77 if (modes == nullptr || mode_count == 0) [[unlikely]] {
78 SDL_free(modes);
79 return false;
80 }
81
82 const SDL_DisplayMode* best = nullptr;
83 SDL_DisplayMode best_mode{};
84 for (int index = 0; index < mode_count; ++index) {
85 const SDL_DisplayMode& mode = *modes[index];
86 if (static_cast<uint32_t>(mode.w) != width ||
87 static_cast<uint32_t>(mode.h) != height) {
88 continue;
89 }
90
91 const uint32_t hz = RefreshRateToHz(mode);
92 if (refresh_rate != 0U && hz == refresh_rate) {
93 out_mode = mode;
94 SDL_free(modes);
95 return true;
96 }
97
98 if (best == nullptr) {
99 best_mode = mode;
100 best = &best_mode;
101 }
102 }
103
104 if (best != nullptr) {
105 out_mode = *best;
106 SDL_free(modes);
107 return true;
108 }
109
110 SDL_free(modes);
111 return false;
112}

427void ApplyWindowIcons(SDL_Window& native_window,
428 std::span<const ::helios::window::IconImage> icons) {
429 if (icons.empty()) {
430 SDL_SetWindowIcon(&native_window, nullptr);
431 return;
432 }
433
434 std::vector<SDL_Surface*> surfaces;
435 surfaces.reserve(icons.size());
436 for (const ::helios::window::IconImage& icon : icons) {
437 if (icon.width == 0 || icon.height == 0 ||
438 icon.rgba.size() != static_cast<size_t>(icon.width) *
439 static_cast<size_t>(icon.height) * 4U) {
440 continue;
441 }
442
443 SDL_Surface* surface = SDL_CreateSurfaceFrom(
444 static_cast<int>(icon.width), static_cast<int>(icon.height),
445 SDL_PIXELFORMAT_RGBA32, const_cast<uint8_t*>(icon.rgba.data()),
446 static_cast<int>(icon.width) * 4);
447 if (surface != nullptr) {
448 surfaces.push_back(surface);
449 }
450 }
451
452 if (!surfaces.empty()) {
453 SDL_SetWindowIcon(&native_window, surfaces.front());
454 }
455
456 for (SDL_Surface* surface : surfaces) {
457 SDL_DestroySurface(surface);
458 }
459}

562 if (clipboard == nullptr) [[unlikely]] {
563 return;
564 }
565
566 if (clipboard->pending_write) {
567 SDL_SetClipboardText(clipboard->text.c_str());
568 clipboard->pending_write = false;
569#ifdef HELIOS_PLATFORM_WINDOWS
570 CaptureClipboardSequence(context);
571#endif
572 context.clipboard_dirty = false;
573 }
574
575 if (native.entries.empty()) [[unlikely]] {
576 context.clipboard_dirty = false;
577 return;
578 }
579

581#ifdef HELIOS_PLATFORM_WINDOWS
582 if (!context.clipboard_sequence_valid ||
583 GetClipboardSequenceNumber() != context.clipboard_sequence) {
584 should_read = true;

586#endif
587
588 if (!should_read) {

590 }
591
592#ifdef HELIOS_PLATFORM_WINDOWS

594 CaptureClipboardSequence(context);
595 context.clipboard_dirty = false;
596 return;
597 }
598#endif

603#endif

607 return;
608 }

611 SDL_free(clipboard_text);
612
613 if (new_text == clipboard->text) {
614 return;
