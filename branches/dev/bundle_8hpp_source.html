264template <typename T>
265constexpr auto FlattenValue(T&& value) {
266 using Decayed = std::remove_cvref_t<T>;
267 if constexpr (IsComponentBundleTypes<Decayed>::value) {
268 return FlattenBundleValues(std::forward<T>(value));
269 } else if constexpr (kIsNestedBundleType<Decayed>) {

272 return FlattenBundleValues(std::move(copy).Build());
