112struct GamepadAxisFilter {

907/**
908 * @brief Formats gamepad dirty flags as a pipe-separated list and writes to an
909 * output iterator.
910 * @tparam It Output iterator type
911 * @param out Output iterator to write the formatted string to
912 * @param flags Combined dirty flags
913 * @param with_prefix Whether to include a "GamepadDirtyFlags::" prefix
914 * @return Updated output iterator after writing the formatted string
915 */
916template <typename It>
917 requires std::output_iterator<It, char>
918inline It ToString(It out, GamepadDirtyFlags flags, bool with_prefix = false) {
919 const std::string_view kNoneStr =
920 with_prefix ? "GamepadDirtyFlags::None" : "None";
921
922 if (flags == GamepadDirtyFlags::kNone) {
923 return std::format_to(out, "{}", kNoneStr);
924 }
925
926 bool first = true;
927 const auto append = [flags, &out, &first, with_prefix](
928 std::string_view name, GamepadDirtyFlags flag) {
929 if (!HasFlag(flags, flag)) {
930 return;
931 }
932 if (!first) {
933 out = std::format_to(out, " | ");
934 }
935 first = false;
936 out = with_prefix ? std::format_to(out, "GamepadDirtyFlags::{}", name)
937 : std::format_to(out, "{}", name);
938 };
939
940 append("Rumble", GamepadDirtyFlags::kRumble);
941 append("TriggerRumble", GamepadDirtyFlags::kTriggerRumble);
942 append("Led", GamepadDirtyFlags::kLed);
943 append("Sensors", GamepadDirtyFlags::kSensors);
944 return out;
945}
