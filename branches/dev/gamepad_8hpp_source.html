89/**
90 * @brief Per-slot rest-center bookkeeping for `UpdateGamepadState`.
91 * @details `raw` stores the last backend samples (GLFW range). `center` is the
92 * calibrated rest for each axis (sticks default `0`, triggers `-1`).
93 */
94struct GamepadAxisFilter {
95 static constexpr size_t kSize = Axis<GamepadAxis>::kSize;
96 static constexpr float kTriggerRest = -1.0F;
97 static constexpr float kRestCapture = 0.35F;
98
99 /// @brief Initializes trigger rest to GLFW `-1`.
100 GamepadAxisFilter() noexcept { Reset(); }
101
102 /// @brief Clears samples and restores default rest centers.
103 void Reset() noexcept;
104
105 Axis<GamepadAxis> raw;
106 std::array<float, kSize> center{};
107 std::array<uint16_t, kSize> rest_frames{};
108 std::array<uint8_t, kSize> seen{};
109};
110
111inline void GamepadAxisFilter::Reset() noexcept {
112 raw.Clear();
113 raw.Set(GamepadAxis::kLeftTrigger, kTriggerRest);
114 raw.Set(GamepadAxis::kRightTrigger, kTriggerRest);
115 center.fill(0.0F);
116 center[static_cast<size_t>(GamepadAxis::kLeftTrigger)] = kTriggerRest;
117 center[static_cast<size_t>(GamepadAxis::kRightTrigger)] = kTriggerRest;
118 rest_frames.fill(0);
119 seen.fill(0);
120}

172/**
173 * @brief Combines gamepad dirty flags.
174 * @param lhs Left-hand side flag
175 * @param rhs Right-hand side flag

178[[nodiscard]] constexpr GamepadDirtyFlags operator|(

642/**
643 * @brief Formats gamepad dirty flags as a pipe-separated list and writes to an
644 * output iterator.
645 * @tparam It Output iterator type
646 * @param flags Combined dirty flags
647 * @param out Output iterator to write the formatted string to
648 * @param with_prefix Whether to include a "GamepadDirtyFlags::" prefix
649 * @return Updated output iterator after writing the formatted string
650 */
651template <typename It>
652 requires std::output_iterator<It, char>
653inline It ToString(GamepadDirtyFlags flags, It out, bool with_prefix = false) {
654 const std::string_view kNoneStr =
655 with_prefix ? "GamepadDirtyFlags::None" : "None";
656
657 if (flags == GamepadDirtyFlags::kNone) {
658 return std::format_to(out, "{}", kNoneStr);
659 }
660
661 bool first = true;
662 const auto append = [flags, &out, &first, with_prefix](
663 std::string_view name, GamepadDirtyFlags flag) {
664 if (!HasFlag(flags, flag)) {
665 return;
666 }
667 if (!first) {
668 out = std::format_to(out, " | ");
669 }
670 first = false;
671 out = with_prefix ? std::format_to(out, "GamepadDirtyFlags::{}", name)
672 : std::format_to(out, "{}", name);
673 };
674
675 append("Rumble", GamepadDirtyFlags::kRumble);
676 append("TriggerRumble", GamepadDirtyFlags::kTriggerRumble);
677 append("Led", GamepadDirtyFlags::kLed);
678 append("Sensors", GamepadDirtyFlags::kSensors);
679 return out;
680}
