76/**
77 * @brief Per-slot rest-center bookkeeping for `UpdateGamepadState`.
78 * @details `raw` stores the last backend samples (GLFW range). `center` is the

81struct GamepadAxisFilter {
82 static constexpr size_t kSize = Axis<GamepadAxis>::kSize;
83 static constexpr float kTriggerRest = -1.0F;
84 static constexpr float kRestCapture = 0.35F;
85
86 /// @brief Initializes trigger rest to GLFW `-1`.
87 GamepadAxisFilter() noexcept { Reset(); }
88
89 /// @brief Clears samples and restores default rest centers.
90 void Reset() noexcept;
91
92 Axis<GamepadAxis> raw;
93 std::array<float, kSize> center{};
94 std::array<uint16_t, kSize> rest_frames{};
95 std::array<uint8_t, kSize> seen{};
96};
97
98inline void GamepadAxisFilter::Reset() noexcept {
99 raw.Clear();
100 raw.Set(GamepadAxis::kLeftTrigger, kTriggerRest);
101 raw.Set(GamepadAxis::kRightTrigger, kTriggerRest);
102 center.fill(0.0F);
103 center[static_cast<size_t>(GamepadAxis::kLeftTrigger)] = kTriggerRest;
104 center[static_cast<size_t>(GamepadAxis::kRightTrigger)] = kTriggerRest;
105 rest_frames.fill(0);
106 seen.fill(0);
107}
