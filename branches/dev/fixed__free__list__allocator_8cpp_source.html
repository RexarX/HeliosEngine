116void FixedFreeListAllocator::InsertAndCoalesce(FreeBlock* block) noexcept {
117 FreeBlock* previous = nullptr;
118 FreeBlock* current = free_list_;
119 while (current != nullptr && current < block) {
120 previous = current;
121 current = current->next;
122 }
123
124 block->next = current;
125
126 if (previous == nullptr) {
127 free_list_ = block;
128 } else {
129 previous->next = block;
130 }
131
132 free_block_count_.fetch_add(1, std::memory_order_relaxed);
133 if (current != nullptr && reinterpret_cast<std::byte*>(block) + block->size ==
134 reinterpret_cast<std::byte*>(current)) {
135 block->size += current->size;
136 block->next = current->next;
137 free_block_count_.fetch_sub(1, std::memory_order_relaxed);
138 }
139
140 if (previous != nullptr &&
141 reinterpret_cast<std::byte*>(previous) + previous->size ==
142 reinterpret_cast<std::byte*>(block)) {
143 previous->size += block->size;
144 previous->next = block->next;
145 free_block_count_.fetch_sub(1, std::memory_order_relaxed);
146 }

163void* FixedFreeListAllocator::do_allocate(size_t bytes, size_t alignment) {
164 HELIOS_MEMORY_PROFILE_SCOPE_N(
165 "helios::mem::FixedFreeListAllocator::do_allocate");
166 HELIOS_MEMORY_PROFILE_ZONE_VALUE(bytes);
167
168 if (bytes == 0) [[unlikely]] {
169 return nullptr;
170 }
171
172 HELIOS_VERIFY(IsPowerOfTwo(alignment),
173 "alignment must be a power of two, got {}!", alignment);
174
175 const std::scoped_lock lock(mutex_);
176 HELIOS_MEMORY_PROFILE_LOCK_MARK(mutex_);
177
178 const size_t effective_alignment = std::max(alignment, kMinAlignment);
179 FreeBlock* best = nullptr;
180 FreeBlock* best_previous = nullptr;
181 size_t best_padding = 0;
182 size_t best_total = std::numeric_limits<size_t>::max();
183 FreeBlock* previous = nullptr;
184 for (FreeBlock* block = free_list_; block != nullptr;
185 previous = block, block = block->next) {
186 const size_t padding = CalculatePaddingWithHeader(
187 block, effective_alignment, sizeof(AllocationHeader));
188 const size_t total = SaturatingAdd(padding, bytes);
189 if (block->size >= total && total < best_total) {
190 best = block;
191 best_previous = previous;
192 best_padding = padding;
193 best_total = total;
194 }
195 }
196
197 if (best == nullptr) [[unlikely]] {
198 return nullptr;
199 }
200
201 if (best_previous != nullptr) {
202 best_previous->next = best->next;
203 } else {
204 free_list_ = best->next;
205 }
206
207 free_block_count_.fetch_sub(1, std::memory_order_relaxed);
208 size_t remaining = best->size - best_total;
209 auto* split_address = reinterpret_cast<std::byte*>(best) + best_total;
210 const size_t split_padding =
211 CalculatePadding(split_address, alignof(FreeBlock));
212
213 if (remaining >= split_padding + sizeof(FreeBlock) + 1) {
214 best_total += split_padding;
215 remaining -= split_padding;
216 split_address += split_padding;
217 auto* const split =
218 std::launder(reinterpret_cast<FreeBlock*>(split_address));
219 split->size = remaining;
220 split->next = nullptr;
221 InsertAndCoalesce(split);
222 } else {
223 best_total = best->size;
224 }
225
226 auto* const result =
227 std::launder(reinterpret_cast<std::byte*>(best) + best_padding);
228 auto* const header =
229 std::launder(reinterpret_cast<AllocationHeader*>(result) - 1);
230 header->size = best_total;
231 header->padding = best_padding;
232
233 const size_t used =
234 used_memory_.fetch_add(best_total, std::memory_order_relaxed) +
235 best_total;
236 size_t peak = peak_usage_.load(std::memory_order_relaxed);
237 while (used > peak && !peak_usage_.compare_exchange_weak(
238 peak, used, std::memory_order_relaxed)) {
239 }
240
241 allocation_count_.fetch_add(1, std::memory_order_relaxed);
242 total_allocations_.fetch_add(1, std::memory_order_relaxed);
243 alignment_waste_.fetch_add(best_padding - sizeof(AllocationHeader),
244 std::memory_order_relaxed);
245 return result;
246}

249 size_t /*alignment*/) {
250 HELIOS_MEMORY_PROFILE_SCOPE_N(
251 "helios::mem::FixedFreeListAllocator::do_deallocate");
252
253 if (ptr == nullptr) [[unlikely]] {
254 return;
255 }
256
257 HELIOS_ASSERT(Owns(ptr), "ptr does not belong to fixed free-list!");
258 const std::scoped_lock lock(mutex_);
259 HELIOS_MEMORY_PROFILE_LOCK_MARK(mutex_);
260
261 auto* const header =
262 std::launder(reinterpret_cast<AllocationHeader*>(ptr) - 1);
263 auto* const block = std::launder(reinterpret_cast<FreeBlock*>(
264 static_cast<std::byte*>(ptr) - header->padding));
265 block->size = header->size;
266 block->next = nullptr;
267
268 used_memory_.fetch_sub(header->size, std::memory_order_relaxed);
269 allocation_count_.fetch_sub(1, std::memory_order_relaxed);
270 total_deallocations_.fetch_add(1, std::memory_order_relaxed);
271
272 InsertAndCoalesce(block);
