66/// @brief Snapshot of one pen / stylus slot.
67struct Pen {
68 /// @brief Clears identity, buttons, axes, position, and tip state.
69 void Reset() noexcept;
70
71 double position_x = 0.0;
72 double position_y = 0.0;
73 double delta_x = 0.0;
74 double delta_y = 0.0;
75 Axis<PenAxis> axes;
76 PenButtonInput buttons;
77 std::optional<PenId> id;
78 PenDeviceType device_type = PenDeviceType::kUnknown;
79 bool in_proximity = false;
80 bool down = false;
81 bool eraser = false;
82 bool has_position = false;
83
84 /**
85 * @brief Returns the pen position.
86 * @return Window-relative x and y
87 */
88 [[nodiscard]] constexpr auto GetPosition() const noexcept
89 -> std::pair<double, double> {
90 return {position_x, position_y};
91 }
92
93 /**
94 * @brief Returns the per-frame motion delta.
95 * @return Delta x and y
96 */
97 [[nodiscard]] constexpr auto GetDelta() const noexcept
98 -> std::pair<double, double> {
99 return {delta_x, delta_y};
100 }
101};

103inline void Pen::Reset() noexcept {
104 position_x = 0.0;
105 position_y = 0.0;
106 delta_x = 0.0;
107 delta_y = 0.0;
108 axes.Clear();
109 buttons.Reset();
110 id.reset();
111 device_type = PenDeviceType::kUnknown;
112 in_proximity = false;
113 down = false;
114 eraser = false;
115 has_position = false;
116}
