29/// @brief How `FrameLimiter` chooses a frame interval.
30enum class FrameLimiterMode : uint8_t {
31 kOff, ///< Do not sleep.
32 kManual, ///< Sleep using `Interval()`.
33 kAuto, ///< Sleep using `RefreshRate()` when it is non-zero.

36/// @brief Construction / install configuration for `FrameLimiter`.
37struct FrameLimiterSettings {
38 static constexpr auto kDefaultInterval =
39 std::chrono::nanoseconds{16'666'667}; // ~60 FPS
40
41 FrameLimiterMode mode = FrameLimiterMode::kManual;
42 std::chrono::nanoseconds interval = kDefaultInterval;
43
44 /**
45 * @brief Disables pacing.
46 * @return Settings with `kOff`
47 */
48 [[nodiscard]] static constexpr FrameLimiterSettings Off() noexcept {
49 return {.mode = FrameLimiterMode::kOff};
50 }
51
52 /**
53 * @brief Manual cap at the given frames per second.
54 * @param fps Target frames per second (must be > 0)
55 * @return Manual settings with the corresponding interval
56 */
57 [[nodiscard]] static constexpr FrameLimiterSettings FromFPS(
58 size_t fps) noexcept {
59 constexpr size_t kNanosecsInSec = 1'000'000'000;
60 return {.mode = FrameLimiterMode::kManual,
61 .interval = std::chrono::nanoseconds{kNanosecsInSec / fps}};
62 }
63
64 /**
65 * @brief Manual cap at the given frequency in Hz.
66 * @param hz Target frequency in Hz (must be > 0)
67 * @return Manual settings with the corresponding interval
68 */
69 [[nodiscard]] static constexpr FrameLimiterSettings FromHz(
70 double hz) noexcept {
71 const auto nanosec = std::chrono::duration_cast<std::chrono::nanoseconds>(
72 std::chrono::duration<double>{1.0 / hz});
73 return {.mode = FrameLimiterMode::kManual, .interval = nanosec};
74 }
75
76 /**
77 * @brief Manual cap with an explicit interval.
78 * @tparam Rep Duration representation type
79 * @tparam Period Duration period type
80 * @param interval Desired frame interval
81 * @return Manual settings with `interval`
82 */
83 template <typename Rep, typename Period>
84 [[nodiscard]] static constexpr FrameLimiterSettings FromInterval(
85 std::chrono::duration<Rep, Period> interval) noexcept {
86 return {.mode = FrameLimiterMode::kManual,
87 .interval =
88 std::chrono::duration_cast<std::chrono::nanoseconds>(interval)};
89 }
90
91 /**
92 * @brief Match display refresh when `FrameLimiter::RefreshRate()` is set.
93 * @details Treats a refresh rate of `0` as `kOff` until something (usually
94 * the window plugin) reports a monitor Hz.
95 * @return Auto settings
96 */
97 [[nodiscard]] static constexpr FrameLimiterSettings Auto() noexcept {
98 return {.mode = FrameLimiterMode::kAuto};
99 }
100};

118/// @brief Per-world frame pacer. One instance per `App` / `SubApp` world.
119class FrameLimiter {
120public:
121 using Clock = std::chrono::steady_clock;
122
123 static constexpr std::string_view kName = "helios::app::FrameLimiter";
124
125 /**
126 * @brief Constructs a limiter from install settings.
127 * @param settings Mode and interval
128 */
129 explicit FrameLimiter(FrameLimiterSettings settings = {}) noexcept
130 : mode_(settings.mode), interval_ns_(settings.interval.count()) {}
131 FrameLimiter(const FrameLimiter& other) noexcept;
132 FrameLimiter(FrameLimiter&& other) noexcept;
133 ~FrameLimiter() = default;
134
135 FrameLimiter& operator=(const FrameLimiter& other) noexcept;
136 FrameLimiter& operator=(FrameLimiter&& other) noexcept;
137
138 /**
139 * @brief Blocks until the next tick boundary when pacing is active.
140 * @details Not thread-safe against itself: call only from the thread that
141 * drives this world's `Update()`. The first call after a deadline reset does
142 * not sleep. `kOff`, a non-positive interval, and `kAuto` with a zero
143 * refresh rate clear the deadline and return immediately.
144 */
145 void Wait();
146
147 /**
148 * @brief Sets the limiter mode. Thread-safe.
149 * @param mode New mode
150 */
151 void SetMode(FrameLimiterMode mode) noexcept {
152 mode_.store(mode, std::memory_order_relaxed);
153 }
154
155 /**
156 * @brief Sets the manual interval. Thread-safe.
157 * @param interval Frame interval used by `kManual`
158 */
159 void SetInterval(std::chrono::nanoseconds interval) noexcept {
160 interval_ns_.store(interval.count(), std::memory_order_relaxed);
161 }
162
163 /**
164 * @brief Sets the display refresh rate used by `kAuto`. Thread-safe.
165 * @param hz Refresh rate in Hertz; `0` means unknown
166 */
167 void SetRefreshRate(uint32_t hz) noexcept {
168 refresh_rate_hz_.store(hz, std::memory_order_relaxed);
169 }
170
171 /**
172 * @brief Returns the current mode. Thread-safe.
173 * @return Current `FrameLimiterMode`
174 */
175 [[nodiscard]] FrameLimiterMode Mode() const noexcept {
176 return mode_.load(std::memory_order_relaxed);
177 }
178
179 /**
180 * @brief Returns the manual interval. Thread-safe.
181 * @return Manual frame interval
182 */
183 [[nodiscard]] std::chrono::nanoseconds Interval() const noexcept {
184 return std::chrono::nanoseconds{
185 interval_ns_.load(std::memory_order_relaxed)};
186 }
187
188 /**
189 * @brief Returns the Auto-mode refresh rate. Thread-safe.
190 * @return Hertz, or `0` if unknown
191 */
192 [[nodiscard]] uint32_t RefreshRate() const noexcept {
193 return refresh_rate_hz_.load(std::memory_order_relaxed);
194 }
195
196private:
197 [[nodiscard]] auto ResolveInterval() const noexcept
198 -> std::optional<std::chrono::nanoseconds>;
199
200 std::atomic<FrameLimiterMode> mode_{FrameLimiterMode::kManual};
201 std::atomic<int64_t> interval_ns_{
202 FrameLimiterSettings::kDefaultInterval.count()};
203 std::atomic<uint32_t> refresh_rate_hz_{0};
204 std::optional<Clock::time_point> next_tick_;
205};
