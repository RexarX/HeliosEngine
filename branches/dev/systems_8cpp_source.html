58void CaptureRestIfFirstSample(GamepadAxisFilter& filter, GamepadAxis axis,
59 float raw, bool auto_calibrate) noexcept {
60 const size_t index = AxisIndex(axis);
61 if (filter.seen[index] != 0) {
62 return;
63 }
64 filter.seen[index] = 1;
65 if (!auto_calibrate) {
66 return;
67 }
68
69 const float expected = Trigger(axis) ? GamepadAxisFilter::kTriggerRest : 0.0F;
70 if (std::fabs(raw - expected) <= GamepadAxisFilter::kRestCapture) {
71 filter.center[index] = raw;
72 }
73}

75void RecenterAxis(GamepadAxisFilter& filter, GamepadAxis axis, bool resting,
76 uint16_t rest_frames, bool auto_calibrate) noexcept {
77 const size_t index = AxisIndex(axis);
78 if (!auto_calibrate) {
79 filter.rest_frames[index] = 0;
80 return;
81 }
82
83 if (!resting) {
84 filter.rest_frames[index] = 0;
85 return;
86 }
87
88 if (filter.rest_frames[index] < rest_frames) {
89 ++filter.rest_frames[index];
90 }
91 if (filter.rest_frames[index] >= rest_frames) {
92 filter.center[index] = filter.raw.Get(axis);
93 filter.rest_frames[index] = 0;
94 }

97void RecenterStick(GamepadAxisFilter& filter, GamepadAxis x_axis,
98 GamepadAxis y_axis, const AxisFilter& stick,
99 uint16_t rest_frames, bool auto_calibrate) noexcept {
100 const float dx = filter.raw.Get(x_axis) - filter.center[AxisIndex(x_axis)];
101 const float dy = filter.raw.Get(y_axis) - filter.center[AxisIndex(y_axis)];
102 const bool resting = std::hypot(dx, dy) < stick.deadzone;
103 RecenterAxis(filter, x_axis, resting, rest_frames, auto_calibrate);
104 RecenterAxis(filter, y_axis, resting, rest_frames, auto_calibrate);
105}
