39void CaptureRestIfFirstSample(GamepadAxisFilter& filter, GamepadAxis axis,
40 float raw, bool auto_calibrate) noexcept {
41 const size_t index = AxisIndex(axis);
42 if (filter.seen[index] != 0) {
43 return;
44 }
45 filter.seen[index] = 1;
46 if (!auto_calibrate) {
47 return;
48 }
49
50 const float expected = Trigger(axis) ? GamepadAxisFilter::kTriggerRest : 0.0F;
51 if (std::fabs(raw - expected) <= GamepadAxisFilter::kRestCapture) {
52 filter.center[index] = raw;
53 }
54}

56void RecenterAxis(GamepadAxisFilter& filter, GamepadAxis axis, bool resting,
57 uint16_t rest_frames, bool auto_calibrate) noexcept {
58 const size_t index = AxisIndex(axis);
59 if (!auto_calibrate) {
60 filter.rest_frames[index] = 0;
61 return;
62 }
63
64 if (!resting) {
65 filter.rest_frames[index] = 0;
66 return;
67 }
68
69 if (filter.rest_frames[index] < rest_frames) {
70 ++filter.rest_frames[index];
71 }
72 if (filter.rest_frames[index] >= rest_frames) {
73 filter.center[index] = filter.raw.Get(axis);
74 filter.rest_frames[index] = 0;
75 }
76}

78void RecenterStick(GamepadAxisFilter& filter, GamepadAxis x_axis,
79 GamepadAxis y_axis, const AxisFilter& stick,
80 uint16_t rest_frames, bool auto_calibrate) noexcept {
81 const float dx = filter.raw.Get(x_axis) - filter.center[AxisIndex(x_axis)];
82 const float dy = filter.raw.Get(y_axis) - filter.center[AxisIndex(y_axis)];
83 const bool resting = std::hypot(dx, dy) < stick.deadzone;
84 RecenterAxis(filter, x_axis, resting, rest_frames, auto_calibrate);
85 RecenterAxis(filter, y_axis, resting, rest_frames, auto_calibrate);
