49auto DynamicLibrary::Load(const std::filesystem::path& path)
50 -> DynamicLibraryResult<void> {
51 if (Loaded()) {
52 return std::unexpected(DynamicLibraryError::kAlreadyLoaded);
53 }
54
55 if (!std::filesystem::exists(path)) {
56 return std::unexpected(DynamicLibraryError::kFileNotFound);
57 }

70 }

72 path_ = path;
73 return {};

115auto DynamicLibrary::GetSymbolAddress(std::string_view name) const
116 -> DynamicLibraryResult<void*> {
117 if (!Loaded()) {
