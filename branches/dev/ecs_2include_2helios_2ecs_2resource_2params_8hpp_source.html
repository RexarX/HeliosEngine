24/**

33public:

86/**
87 * @brief Read-only or mutable access to a system-local resource.
88 * @details Only one system ever accesses it.
89 * Use `Local<const T>` for read-only access and `Local<T>` for mutable access.
90 * @tparam T Resource type. Qualify with `const` for read-only access
91 */
92template <ResourceTrait T>
93class Local {
94public:
95 using DecayedT = std::remove_cvref_t<T>;
96
97 /**

99 * @param resource Resource value
100 */
101 explicit constexpr Local(T& resource) noexcept : resource_(resource) {}
102 constexpr Local(const Local&) noexcept = default;
103 constexpr Local(Local&&) noexcept = default;
104 constexpr ~Local() noexcept = default;
105
106 constexpr Local& operator=(const Local&) noexcept = default;
107 constexpr Local& operator=(Local&&) noexcept = default;

109 /**
110 * @brief Assigns a new value to the local resource.
111 * @tparam U Resource type
112 * @param value Resource value
113 * @return Reference to the assigned resource
114 */
115 template <ResourceTrait U>
116 requires std::same_as<DecayedT, std::remove_cvref_t<U>> &&
117 std::is_assignable_v<T&, U> && (!std::is_const_v<T>)
118 constexpr T& operator=(U&& value) const
119 noexcept(std::is_nothrow_assignable_v<T&, U>) {
120 resource_.get() = std::forward<U>(value);
121 return resource_.get();
122 }
123
124 [[nodiscard]] constexpr T& operator*() const noexcept {
125 return resource_.get();
126 }

130 }

134 * @return Reference to the local resource

138private:
