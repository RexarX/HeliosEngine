427 * @param lhs Left-hand side modifiers
428 * @param rhs Right-hand side modifiers

431[[nodiscard]] constexpr Modifiers operator|(Modifiers lhs,

433 return static_cast<Modifiers>(std::to_underlying(lhs) |

437/**
438 * @brief Tests whether a modifier flag is set.
439 * @param modifiers Combined modifier flags
440 * @param flag Flag to test
441 * @return True if the flag is set, false otherwise
442 */
443[[nodiscard]] constexpr bool HasFlag(Modifiers modifiers,
444 Modifiers flag) noexcept {
445 return (std::to_underlying(modifiers) & std::to_underlying(flag)) != 0U;
446}

448/**
449 * @brief Formats modifiers as a pipe-separated list and writes to an output
450 * iterator.
451 * @tparam It Output iterator type
452 * @param out Output iterator to write the formatted string to
453 * @param modifiers Combined modifier flags
454 * @param with_prefix Whether to include a "Modifiers::" prefix for each flag
455 * @return Updated output iterator after writing the formatted string
456 */
457template <typename It>
458 requires std::output_iterator<It, char>
459inline It ToString(It out, Modifiers modifiers, bool with_prefix = false) {
460 const std::string_view kNoneStr = with_prefix ? "Modifiers::None" : "None";
461
462 if (modifiers == Modifiers::kNone) {
463 return std::format_to(out, "{}", kNoneStr);
464 }
465
466 bool first = true;
467 const auto append = [modifiers, &out, &first, with_prefix](
468 std::string_view name, Modifiers flag) {
469 if (!HasFlag(modifiers, flag)) {
470 return;
471 }
472 if (!first) {
473 out = std::format_to(out, " | ");
474 }
475 first = false;
476 out = with_prefix ? std::format_to(out, "Modifiers::{}", name)
477 : std::format_to(out, "{}", name);
478 };
479
480 append("Shift", Modifiers::kShift);
481 append("Control", Modifiers::kControl);
482 append("Alt", Modifiers::kAlt);
483 append("Super", Modifiers::kSuper);
484 append("CapsLock", Modifiers::kCapsLock);
485 append("NumLock", Modifiers::kNumLock);
486 return out;
487}

489/**
490 * @brief Formats modifiers as a pipe-separated list of flag names.
491 * @param modifiers Combined modifier flags
492 * @param with_prefix Whether to include a "Modifiers::" prefix for each flag
493 * @return Formatted string
494 */
495[[nodiscard]] inline std::string ToString(Modifiers modifiers,
496 bool with_prefix = false) {
497 std::string result;
