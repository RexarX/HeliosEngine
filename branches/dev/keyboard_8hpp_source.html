522/**
523 * @brief Combines modifier flags.
524 * @param lhs Left-hand side modifiers
525 * @param rhs Right-hand side modifiers
526 * @return Combined modifiers
527 */
528[[nodiscard]] constexpr Modifiers operator|(Modifiers lhs,
529 Modifiers rhs) noexcept {
530 return static_cast<Modifiers>(std::to_underlying(lhs) |
531 std::to_underlying(rhs));
532}

534/**
535 * @brief Tests whether a modifier flag is set.
536 * @param modifiers Combined modifier flags
537 * @param flag Flag to test
538 * @return True if the flag is set, false otherwise
539 */
540[[nodiscard]] constexpr bool HasFlag(Modifiers modifiers,
541 Modifiers flag) noexcept {
542 return (std::to_underlying(modifiers) & std::to_underlying(flag)) != 0U;
543}

545/**
546 * @brief Formats modifiers as a pipe-separated list and writes to an output
547 * iterator.
548 * @tparam It Output iterator type
549 * @param out Output iterator to write the formatted string to
550 * @param modifiers Combined modifier flags
551 * @param with_prefix Whether to include a "Modifiers::" prefix for each flag
552 * @return Updated output iterator after writing the formatted string
553 */
554template <typename It>
555 requires std::output_iterator<It, char>
556inline It ToString(It out, Modifiers modifiers, bool with_prefix = false) {
557 const std::string_view kNoneStr = with_prefix ? "Modifiers::None" : "None";
558
559 if (modifiers == Modifiers::kNone) {
560 return std::format_to(out, "{}", kNoneStr);
561 }
562
563 bool first = true;
564 const auto append = [modifiers, &out, &first, with_prefix](
565 std::string_view name, Modifiers flag) {
566 if (!HasFlag(modifiers, flag)) {
567 return;
568 }
569 if (!first) {
570 out = std::format_to(out, " | ");
571 }
572 first = false;
573 out = with_prefix ? std::format_to(out, "Modifiers::{}", name)
574 : std::format_to(out, "{}", name);
575 };
576
577 append("Shift", Modifiers::kShift);
578 append("Control", Modifiers::kControl);
579 append("Alt", Modifiers::kAlt);
580 append("Super", Modifiers::kSuper);
581 append("CapsLock", Modifiers::kCapsLock);
582 append("NumLock", Modifiers::kNumLock);
583 return out;
584}
