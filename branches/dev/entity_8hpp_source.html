10/**
11 * @brief Unique identifier for entities with generation counter to handle
12 * recycling.
13 * @details Entity uses a combination of index and generation to provide stable
14 * references even when entities are destroyed and their indices are recycled.
15 *
16 * Generations encode both a 31-bit reuse counter and a liveness bit:
17 * - bit 31 (`kAliveBit`) is set when the slot is living;
18 * - bits 0–30 hold the counter.
19 *
20 * Memory layout: 32-bit index + 32-bit generation = 64-bit total
21 *
22 * @note Individual Entity value objects are safe to copy freely. Concurrent
23 * mutation of the same Entity instance is not synchronized.
24 */
25class Entity {
26public:
27 using IndexType = uint32_t;
28 using GenerationType = uint32_t;
29
30 static constexpr auto kInvalidIndex = std::numeric_limits<IndexType>::max();
31 /// Unreachable sentinel: counter mask with the alive bit set.
32 static constexpr auto kInvalidGeneration =
33 std::numeric_limits<GenerationType>::max();
34 /// Top bit: set when the generation represents a living entity.
35 static constexpr GenerationType kAliveBit = 0x8000'0000U;
36 /// Low 31 bits: reuse counter. `kCounterMask` itself is reserved so that
37 /// `kInvalidGeneration` (`kCounterMask | kAliveBit`) stays unreachable.
38 static constexpr GenerationType kCounterMask = 0x7FFF'FFFFU;
39 /// Initial alive generation assigned to brand-new entity indices.
40 static constexpr GenerationType kInitialAliveGeneration = kAliveBit | 1U;
41
42 /**
43 * @brief Constructs an invalid entity.
44 * @details Creates an entity with invalid index and generation values.
45 */
46 constexpr Entity() noexcept = default;
47
48 /**
49 * @brief Constructs entity with specific index and generation.
50 * @param index The entity index
51 * @param generation The entity generation
52 */
53 constexpr Entity(IndexType index, GenerationType generation) noexcept
54 : index_(index), generation_(generation) {}
55 constexpr Entity(const Entity&) noexcept = default;
56 constexpr Entity(Entity&&) noexcept = default;
57 constexpr ~Entity() noexcept = default;

66 /**
67 * @brief Checks if the entity handle is structurally valid.
68 * @details An entity is structurally valid if both its index and generation
69 * are not the reserved invalid sentinels. This does **not** mean the entity
70 * is currently alive in an `EntityManager` — use
71 * `EntityManager::Validate()` for that.
72 * @return True if entity has usable index and generation fields
73 */
74 [[nodiscard]] constexpr bool Valid() const noexcept {
75 return index_ != kInvalidIndex && generation_ != kInvalidGeneration;
76 }
77
78 /**
79 * @brief Checks whether the generation encodes a living entity.
80 * @details Uses the alive bit in the generation encoding. This reflects the
81 * handle's generation value, not whether the entity is currently alive in an
82 * `EntityManager` — use `EntityManager::Validate()` for that.
83 * @return True if generation is alive-encoded and not the invalid sentinel
84 */
85 [[nodiscard]] constexpr bool Alive() const noexcept;

87 /**
88 * @brief Generates a hash value for this entity.
89 * @details Combines index and generation into a 64-bit hash value.
90 * Invalid entities always return hash value of 0.
91 * @return Hash combining generation (high bits) and index (low bits)
92 */
93 [[nodiscard]] constexpr size_t Hash() const noexcept;
94
95 /**
96 * @brief Gets the index component of the entity.
97 * @details The index identifies the entity's storage location in sparse
98 * arrays.
99 * @return Entity index, or `kInvalidIndex` if entity is invalid
100 */
101 [[nodiscard]] constexpr IndexType Index() const noexcept { return index_; }

104 * @brief Gets the generation component of the entity.
105 * @details Returns the bit-encoded generation (alive bit + counter), not a
106 * raw reuse count. Stale handles keep their old encoding after recycling.
107 * @return Entity generation, or `kInvalidGeneration` if entity is invalid
108 */

111 }

113private:

115 GenerationType generation_ =

117};

126constexpr size_t Entity::Hash() const noexcept {
127 if (!Valid()) [[unlikely]] {
128 return 0;
129 }
130
131 auto hash = static_cast<size_t>(index_);
132 hash ^= static_cast<size_t>(generation_) +
133 static_cast<size_t>(0x9e3779b97f4a7c15ULL) + (hash << 6U) +
134 (hash >> 2U);
135 return hash == 0 ? 1 : hash;
136}

138/**
139 * @brief Checks if generation is considered alive.
140 * @details Requires the alive bit and rejects the unreachable invalid
141 * sentinel.
142 * @param gen Generation encoding to check
143 * @return True if generation is alive and not invalid
144 */
145[[nodiscard]] constexpr bool IsAliveGeneration(
146 Entity::GenerationType gen) noexcept {
147 return (gen & Entity::kAliveBit) != 0U && gen != Entity::kInvalidGeneration;
148}

150constexpr bool Entity::Alive() const noexcept {

154/**
155 * @brief Advances a generation toward the requested liveness state.
156 * @details One counter increment occurs per full reuse cycle, on the
157 * alive->free transition. Free->alive only sets the alive bit so a reserved
158 * handle can carry the future alive value while storage still holds the free
159 * encoding (making pre-`Flush` validation fail).
160 *
161 * The reserved counter value `kCounterMask` is skipped so
162 * `kInvalidGeneration` stays unreachable.
163 *
164 * @param gen Current generation encoding
165 * @param alive Target liveness (`true` = free->alive, `false` = alive->free)
166 * @return Next generation encoding for the requested state
167 */
168[[nodiscard]] constexpr Entity::GenerationType NextGeneration(
169 Entity::GenerationType gen, bool alive) noexcept {
170 if (alive) {
171 return (gen & Entity::kCounterMask) | Entity::kAliveBit;
172 }
173
174 auto counter = (gen & Entity::kCounterMask) + 1U;
175 if (counter == Entity::kCounterMask) {
176 counter = 0U;
177 }
178 return counter;
179}
