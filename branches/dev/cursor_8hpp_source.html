13 * @details Retention (previous/current buffers) and delivery (this cursor) are
14 * independent. Two systems each keep their own cursor and can both observe the
15 * same retained message once. Store via `Local` / `SystemLocalData` for
16 * system parameters; pass explicitly to `World::ReadMessages`.
17 * @tparam T Message type satisfying `MessageTrait`
18 */
19template <MessageTrait T>
20struct MessageCursor {
21 /// @brief Next message id this cursor has not yet observed.

23
24 /**
25 * @brief Creates a cursor that includes all currently retained messages.
26 * @return Cursor starting at id 0
27 */
28 [[nodiscard]] static constexpr MessageCursor IncludeBacklog() noexcept {
29 return {.last_message_count = MessageId<T>{0}};
30 }
31

33 * @brief Creates a cursor that ignores currently retained messages.
34 * @param manager Message manager providing the current stream end
35 * @return Cursor positioned at the next id to be written

41
42 /**
43 * @brief Skips all currently retained messages for this cursor.
