25[[nodiscard]] constexpr auto ResolveClientSize(
26 const window::Properties& properties) noexcept -> std::pair<int, int> {
27 const uint32_t width =
28 properties.client_width.value_or(properties.width.value_or(1U));
29 const uint32_t height =
30 properties.client_height.value_or(properties.height.value_or(1U));
31 return {static_cast<int>(width), static_cast<int>(height)};
32}
