26/// @brief Conditions under which closing a window requests application exit.
27enum class ExitTrigger : uint8_t {
28 kNone = 0,
29 kPrimaryClosed = 1U << 0U, ///< A Primary-tagged window was destroyed.
30 kAllWindowsClosed = 1U << 1U, ///< The last native window was destroyed.
31 kAnyWindowClosed = 1U << 2U, ///< Any single window was destroyed.
32};
33
34/**
35 * @brief Combines exit triggers.
36 * @param lhs Left-hand side trigger flags
37 * @param rhs Right-hand side trigger flags
38 * @return Combined trigger flags
39 */
40[[nodiscard]] constexpr ExitTrigger operator|(ExitTrigger lhs,
41 ExitTrigger rhs) noexcept {
42 return static_cast<ExitTrigger>(std::to_underlying(lhs) |
43 std::to_underlying(rhs));
44}
45
46/**
47 * @brief Tests whether an exit trigger flag is set.
48 * @param triggers Combined exit trigger flags
49 * @param trigger Flag to test
50 * @return True if the flag is set, false otherwise
51 */
52[[nodiscard]] constexpr bool HasFlag(ExitTrigger triggers,
53 ExitTrigger trigger) noexcept {
54 return (std::to_underlying(triggers) & std::to_underlying(trigger)) != 0U;
55}

57/// @brief Never auto-exit when windows close.
58inline constexpr auto kExitTriggersNone = ExitTrigger::kNone;
59
60/// @brief Exit when the last window is destroyed.
61inline constexpr auto kExitTriggersLastWindow = ExitTrigger::kAllWindowsClosed;
62
63/// @brief Exit when the primary window is destroyed.
64inline constexpr auto kExitTriggersPrimary = ExitTrigger::kPrimaryClosed;
65
66/// @brief Exit when the primary or last window is destroyed.
67inline constexpr auto kExitTriggersDefault =
68 ExitTrigger::kPrimaryClosed | ExitTrigger::kAllWindowsClosed;

70/**
71 * @brief Determines whether a destroyed window should request application exit.
72 * @param triggers Configured exit triggers
73 * @param was_primary True if the destroyed window had the Primary component
74 * @param is_last_native_window True if no native windows remain after destroy
75 * @return True when an exit trigger matched
76 */
77[[nodiscard]] constexpr bool ShouldRequestExitOnClose(
78 ExitTrigger triggers, bool was_primary,
79 bool is_last_native_window) noexcept {
80 if (HasFlag(triggers, ExitTrigger::kAnyWindowClosed)) {
81 return true;
82 }
83 if (HasFlag(triggers, ExitTrigger::kPrimaryClosed) && was_primary) {
84 return true;
85 }
86 if (HasFlag(triggers, ExitTrigger::kAllWindowsClosed) &&
87 is_last_native_window) {
88 return true;
89 }
90 return false;
91}

113/**
114 * @brief Formats exit triggers as a pipe-separated list and writes to an output
115 * iterator.
116 * @tparam It Output iterator type
117 * @param out Output iterator to write the formatted string to
118 * @param triggers Combined exit trigger flags
119 * @param with_prefix Whether to include an "ExitTrigger::" prefix for each flag
120 * @return Updated output iterator after writing the formatted string
121 */
122template <typename It>
123 requires std::output_iterator<It, char>
124inline It ToString(It out, ExitTrigger triggers, bool with_prefix = false) {
125 const std::string_view kNoneStr = with_prefix ? "ExitTrigger::None" : "None";
126
127 if (triggers == ExitTrigger::kNone) {
128 return std::format_to(out, "{}", kNoneStr);
129 }
130
131 bool first = true;
132 const auto append = [triggers, &out, &first, with_prefix](
133 std::string_view name, ExitTrigger flag) {
134 if (!HasFlag(triggers, flag)) {
135 return;
136 }
137 if (!first) {
138 out = std::format_to(out, " | ");
139 }
140 first = false;
141 out = with_prefix ? std::format_to(out, "ExitTrigger::{}", name)
142 : std::format_to(out, "{}", name);
143 };
144
145 append("PrimaryClosed", ExitTrigger::kPrimaryClosed);
146 append("AllWindowsClosed", ExitTrigger::kAllWindowsClosed);
147 append("AnyWindowClosed", ExitTrigger::kAnyWindowClosed);
148 return out;
149}
