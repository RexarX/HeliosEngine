80Entity EntityManager::Create() {
81 HELIOS_ASSERT(!NeedsFlush(), "Flush reserved entities before creation!");
82
83 // Reuse a free slot if available
84 const int64_t cursor = free_cursor_.load(std::memory_order_relaxed);
85 if (cursor > 0) {
86 const int64_t new_cursor = cursor - 1;
87 // Try to claim the top free slot
88 int64_t expected = cursor;
89 if (free_cursor_.compare_exchange_strong(expected, new_cursor,
90 std::memory_order_relaxed)) {
91 const Entity::IndexType index =
92 free_indices_[static_cast<size_t>(new_cursor)];
93 free_indices_.pop_back();
94 free_cursor_.store(static_cast<int64_t>(free_indices_.size()),
95 std::memory_order_relaxed);
96 const Entity::GenerationType free_gen =
97 GenRef(index).load(std::memory_order_relaxed);
98 return CreateEntityWithId(index,
99 NextGeneration(free_gen, /*alive=*/true));
100 }
101 }
102
103 // No free slot available — allocate a new index
104 const Entity::IndexType index =
105 next_index_.fetch_add(1, std::memory_order_relaxed);
106 return CreateEntityWithId(index, Entity::kInitialAliveGeneration);
107}

109void EntityManager::Destroy(Entity entity) {
110 HELIOS_ASSERT(!NeedsFlush(), "Flush reserved entities before destruction!");
111 HELIOS_ASSERT(entity.Valid(), "Entity '{}' is invalid!", entity);
112
113 if (!Validate(entity)) [[unlikely]] {
114 return;
115 }
116
117 const Entity::IndexType index = entity.Index();
118 GenRef(index).store(NextGeneration(entity.Generation(), /*alive=*/false),
119 std::memory_order_relaxed);
120 free_indices_.push_back(index);
121
122 free_cursor_.store(static_cast<int64_t>(free_indices_.size()),
123 std::memory_order_relaxed);
124 entity_count_.fetch_sub(1, std::memory_order_relaxed);
125}
