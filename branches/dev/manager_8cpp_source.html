83void ComponentManager::Clear(Entity entity) {
84 HELIOS_ASSERT(entity.Valid(), "Entity '{}' is invalid!", entity);
85 const auto it = entity_archetype_.find(entity.Index());

112Archetype& ComponentManager::GetOrCreateArchetype(const ArchetypeId& id) {
113 HELIOS_ECS_PROFILE_SCOPE_N(
114 "helios::ecs::ComponentManager::GetOrCreateArchetype");
115 HELIOS_ECS_PROFILE_ZONE_VALUE(id.Types().size());
116
117 const size_t hash = id.Hash();
118 if (const auto it = archetype_map_.find(hash); it != archetype_map_.end()) {
119 return it->second.archetype.get();
120 }
121
122 // Create archetype in the stable deque storage.
123 archetype_storage_.emplace_back(id);
124 Archetype& archetype = archetype_storage_.back();
125
126 // Create record referencing the archetype.
127 archetype_map_.emplace(hash, ArchetypeRecord{.archetype = std::ref(archetype),
128 .add_edges = {},
129 .remove_edges = {}});
130
131 // Initialize columns with type info from metadata.
132 InitArchetypeColumns(archetype);
133 archetype_list_.emplace_back(archetype);
134 ++structural_version_;
135 return archetype;
136}

152void ComponentManager::MigrateEntity(Entity entity, Archetype& src,
153 Archetype& dst) {
154 HELIOS_ECS_PROFILE_SCOPE_N("helios::ecs::ComponentManager::MigrateEntity");
155 HELIOS_ECS_PROFILE_ZONE_VALUE(src.EntityCount());
156 HELIOS_ECS_PROFILE_ZONE_VALUE(dst.Id().Types().size());
157
158 HELIOS_ASSERT(entity.Valid(), "Entity '{}' is invalid!", entity);
159
160 if (&src == &dst) [[unlikely]] {
161 return;
162 }
163
164 const auto src_row = src.Row(entity);
165
166 // Phase 1: Allocate a row in the destination archetype for this entity.
167 [[maybe_unused]] const auto dst_row = dst.AllocateRow(entity);
168
169 // Phase 2: For each column in dst, either move data from src or push a
170 // default placeholder.
171 for (const auto type_index : dst.Id().Types()) {
172 auto* dst_col = dst.TryColumn(type_index);
173 if (dst_col == nullptr) {
174 continue;
175 }
176
177 const auto meta_it = metadata_.find(type_index);
178 HELIOS_ASSERT(meta_it != metadata_.end(),
179 "Component metadata not registered for a column in target "
180 "archetype!");
181
182 if (src.HasColumn(type_index)) {
183 // Shared column: move the element from src[src_row] into dst (appended
184 // at back).
185 auto* src_col = src.TryColumn(type_index);
186 if (src_col != nullptr && !src_col->Empty() &&
187 meta_it->second.move_column_element != nullptr) {
188 meta_it->second.move_column_element(*dst_col, *src_col,
189 static_cast<size_t>(src_row));
190 }
191 } else {
192 // New column in dst that src doesn't have.
193 // Push a default-constructed placeholder; the caller will overwrite it
194 // via Set.
195 if (meta_it->second.default_push != nullptr) {
196 meta_it->second.default_push(*dst_col);
197 }
198 }
199 }
200
201 // Phase 3: Remove the entity from the source archetype using swap-and-pop.
202 const Entity swapped = src.Remove(entity);
203
204 // Phase 4: Update entity-to-archetype mapping.
205 entity_archetype_.insert_or_assign(entity.Index(), std::ref(dst));
206 if (swapped.Valid()) {
207 entity_archetype_.insert_or_assign(swapped.Index(), std::ref(src));
208 }
209
210 ++structural_version_;
211}
