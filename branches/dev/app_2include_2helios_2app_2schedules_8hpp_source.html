150/**
151 * @brief Registers default application schedules on an ECS scheduler.
152 * @details Creates empty schedules for all built-in types and assigns default
153 * executor kinds, stage membership, and ordering.
154 * @param scheduler ECS scheduler to populate
155 */
156inline void RegisterBuiltinSchedules(ecs::Scheduler& scheduler) {
157 const auto add_in_stage = []<ecs::ScheduleTrait T, ecs::StageTrait S>(
158 ecs::Scheduler& sched, const T& schedule_type,
159 const S& stage, ecs::ExecutorKind kind) {
160 if (sched.TryGetSchedule(schedule_type) != nullptr) {
161 return;
162 }
163 auto schedule = ecs::Schedule::From(schedule_type);
164 schedule.Settings().executor_kind = kind;
165 sched.Add(schedule_type, std::move(schedule)).InStage(stage);
166 };
167
168 if (!scheduler.HasStage(kStartupStage)) {
169 scheduler.AddStage(kStartupStage);
170 }
171
172 // FramePumpOrder ends at Update — advance message buffers there for nested
173 // pumps. MainFrameOrder also includes Update but only advances on its last
174 // present stage (Extract); see Scheduler::RunFrameOrder.
175 if (!scheduler.HasStage(kUpdateStage)) {
176 auto& settings = scheduler.AddStage(kUpdateStage).Settings();
177 settings.advance_messages = true;
178 } else {
179 auto& settings = scheduler.GetStageSettings(kUpdateStage);
180 settings.advance_messages = true;
181 }
182
183 // MainFrameOrder ends at Extract: flush after extract, advance message
184 // buffers once at end of the main frame order.
185 if (!scheduler.HasStage(kExtractStage)) {
186 auto& settings = scheduler.AddStage(kExtractStage).Settings();
187 settings.apply_commands = true;
188 settings.advance_messages = true;
189 } else {
190 auto& settings = scheduler.GetStageSettings(kExtractStage);
191 settings.apply_commands = true;
192 settings.advance_messages = true;
193 }
194
195 if (!scheduler.HasStage(kShutdownStage)) {
196 scheduler.AddStage(kShutdownStage);
197 }
198
199 add_in_stage(scheduler, kMainStartup, kStartupStage,
200 ecs::ExecutorKind::kMainThread);
201 add_in_stage(scheduler, kPreStartup, kStartupStage,
202 ecs::ExecutorKind::kMultiThreaded);
203 add_in_stage(scheduler, kStartup, kStartupStage,
204 ecs::ExecutorKind::kMultiThreaded);
205 add_in_stage(scheduler, kPostStartup, kStartupStage,
206 ecs::ExecutorKind::kMultiThreaded);
207 add_in_stage(scheduler, kFirst, kUpdateStage,
208 ecs::ExecutorKind::kMultiThreaded);
209 add_in_stage(scheduler, kPreUpdate, kUpdateStage,
210 ecs::ExecutorKind::kMultiThreaded);
211 add_in_stage(scheduler, kUpdate, kUpdateStage,
212 ecs::ExecutorKind::kMultiThreaded);
213 add_in_stage(scheduler, kPostUpdate, kUpdateStage,
214 ecs::ExecutorKind::kMultiThreaded);
215 add_in_stage(scheduler, kLast, kUpdateStage,
216 ecs::ExecutorKind::kMultiThreaded);
217 add_in_stage(scheduler, kExtract, kExtractStage,
218 ecs::ExecutorKind::kMultiThreaded);
219 add_in_stage(scheduler, kPreShutdown, kShutdownStage,
220 ecs::ExecutorKind::kMultiThreaded);
221 add_in_stage(scheduler, kShutdown, kShutdownStage,
222 ecs::ExecutorKind::kMainThread);
223 add_in_stage(scheduler, kPostShutdown, kShutdownStage,
224 ecs::ExecutorKind::kMultiThreaded);
225
226 scheduler.Order(kPreStartup).After(kMainStartup);
227 scheduler.Order(kStartup).After(kPreStartup);
228 scheduler.Order(kPostStartup).After(kStartup);
229 scheduler.Order(kPreUpdate).After(kFirst);
230 scheduler.Order(kUpdate).After(kPreUpdate);
231 scheduler.Order(kPostUpdate).After(kUpdate);
232 scheduler.Order(kLast).After(kPostUpdate);
233 scheduler.Order(kShutdown).After(kPreShutdown);
234 scheduler.Order(kPostShutdown).After(kShutdown);
235 scheduler.OrderStage(kExtractStage).After(kUpdateStage);
236}

238/**
239 * @brief Inserts default `MainFrameOrder` and `FramePumpOrder` resources.
240 * @details Safe to call multiple times (`TryInsertResources`).
241 * @param world World that owns the frame-order resources
242 */
243inline void RegisterBuiltinFrameOrders(ecs::World& world) {
244 MainFrameOrder main_order;
245 main_order.TryPushBack(kUpdateStage);
246 main_order.TryPushBack(kExtractStage);
247
248 FramePumpOrder pump_order;
249 pump_order.TryPushBack(kUpdateStage);
250
251 world.TryInsertResources(std::move(main_order), std::move(pump_order));
252}
