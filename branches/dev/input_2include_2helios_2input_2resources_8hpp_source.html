24/// @brief Global input behavior settings.

27/// near-rest sample and recenters after `rest_frames` at rest.
28struct Settings {
29 static constexpr std::string_view kName = "helios::input::Settings";
30 static constexpr uint16_t kDefaultRestFrames = 60;
31
32 AxisFilter stick{.deadzone = AxisFilter::kDefaultDeadzone,
33 .livezone = AxisFilter::kDefaultLivezone,
34 .rescale = true};
35 AxisFilter trigger{.deadzone = AxisFilter::kDefaultTriggerDeadzone,
36 .livezone = AxisFilter::kDefaultLivezone,
37 .rescale = true};
38 uint16_t rest_frames = kDefaultRestFrames;
39 bool auto_calibrate = true;
40 bool raw_mouse_motion = false;
41};
