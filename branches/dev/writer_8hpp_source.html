16/**
17 * @brief Type-safe writer for regular messages.
18 * @details Messages written through `BasicMessageWriter` are buffered in a
19 * message queue and merged into the global `MessageManager` at sync time when
20 * used from system local data. Ids are assigned during that merge.
21 * @note Not thread-safe.
22 * @tparam T Message type satisfying `MessageTrait`
23 */
24template <MessageTrait T>
25class BasicMessageWriter {
26public:
27 /**
28 * @brief Constructs a `BasicMessageWriter` that writes to a message queue.
29 * @param queue Reference to the message queue
30 */
31 explicit constexpr BasicMessageWriter(MessageQueue& queue) noexcept
32 : queue_(queue) {}
33 BasicMessageWriter(const BasicMessageWriter&) = delete;
34 constexpr BasicMessageWriter(BasicMessageWriter&&) noexcept = default;
35 constexpr ~BasicMessageWriter() noexcept = default;
36
37 BasicMessageWriter& operator=(const BasicMessageWriter&) = delete;
38 constexpr BasicMessageWriter& operator=(BasicMessageWriter&&) noexcept =
39 default;
40
41 /**
42 * @brief Writes a single message to the local queue.
43 * @tparam U Message type, must be the same as `T`
44 * @param message Message to write
45 */
46 template <typename U = T>
47 requires std::same_as<std::remove_cvref_t<U>, T>
48 constexpr void Write(U&& message) const {
49 queue_.get().Enqueue(std::forward<U>(message));
50 }
51
52 /**
53 * @brief Writes multiple messages to the local queue in bulk.
54 * @tparam R Range of messages
55 * @param messages Range of messages to write
56 */
57 template <std::ranges::input_range R>
58 requires std::same_as<std::ranges::range_value_t<R>, T>
59 constexpr void WriteBulk(R&& messages) const {
60 queue_.get().EnqueueBulk(std::forward<R>(messages));
61 }
62
63 /**
64 * @brief Constructs an message in-place and writes it to the local queue.
65 * @tparam Args Constructor argument types
66 * @param args Arguments to forward to the message constructor
67 */
68 template <typename... Args>
69 requires std::constructible_from<T, Args...>
70 constexpr void Emplace(Args&&... args) const {
71 queue_.get().Enqueue(T{std::forward<Args>(args)...});
72 }
73
74private:
75 std::reference_wrapper<MessageQueue> queue_; ///< Reference to message queue
76};
