15/**
16 * @brief Type-safe writer for regular messages with a configurable queue
17 * allocator.
18 * @details Messages written through `BasicMessageWriter` are buffered in a
19 * message queue and merged into the global `MessageManager` at sync time when
20 * used from system local data. Ids are assigned during that merge.
21 * @note Not thread-safe.
22 * @tparam T Message type satisfying `MessageTrait`
23 * @tparam Allocator Allocator type for the underlying message queue (default:
24 * `std::allocator<std::byte>`)
25 */
26template <MessageTrait T, typename Allocator = std::allocator<std::byte>>
27class BasicMessageWriter {
28public:
29 /**
30 * @brief Constructs a `BasicMessageWriter` that writes to a message queue.
31 * @param queue Reference to the message queue
32 */
33 explicit constexpr BasicMessageWriter(MessageQueue<Allocator>& queue) noexcept
34 : queue_(queue) {}
35 BasicMessageWriter(const BasicMessageWriter&) = delete;
36 constexpr BasicMessageWriter(BasicMessageWriter&&) noexcept = default;
37 constexpr ~BasicMessageWriter() noexcept = default;
38
39 BasicMessageWriter& operator=(const BasicMessageWriter&) = delete;
40 constexpr BasicMessageWriter& operator=(BasicMessageWriter&&) noexcept =
41 default;
42
43 /**
44 * @brief Writes a single message to the local queue.
45 * @tparam U Message type, must be the same as `T`
46 * @param message Message to write
47 */
48 template <typename U = T>
49 requires std::same_as<std::remove_cvref_t<U>, T>
50 void Write(U&& message) const {
51 queue_.get().Enqueue(std::forward<U>(message));
52 }
53
54 /**
55 * @brief Writes multiple messages to the local queue in bulk.
56 * @tparam R Range of messages
57 * @param messages Range of messages to write
58 */
59 template <std::ranges::input_range R>
60 requires std::same_as<std::ranges::range_value_t<R>, T>
61 void WriteBulk(R&& messages) const {
62 queue_.get().EnqueueBulk(std::forward<R>(messages));
63 }
64
65 /**
66 * @brief Constructs an message in-place and writes it to the local queue.
67 * @tparam Args Constructor argument types
68 * @param args Arguments to forward to the message constructor
69 */
70 template <typename... Args>
71 requires std::constructible_from<T, Args...>
72 void Emplace(Args&&... args) const {
73 queue_.get().Enqueue(T{std::forward<Args>(args)...});
74 }
75
76private:
77 std::reference_wrapper<MessageQueue<Allocator>>
78 queue_; ///< Reference to the message queue
79};

106 /**

108 * @tparam U Message type, must be the same as `T`

112 template <typename U = T>
