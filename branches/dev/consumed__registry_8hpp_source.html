26/**
27 * @brief Per-system registry for tracking consumed message ids.
28 * @details Each system running in parallel gets its own
29 * `ConsumedMessagesRegistry` instance. Systems mark messages as consumed by
30 * writing stable message ids into their registry, avoiding data races. At
31 * Update time, all registries are merged and applied to message queues in
32 * `MessageManager`.
33 *
34 * Entries are monotonic message ids assigned when messages enter the global
35 * manager — not transient indices into the previous/current buffers.
36 *
37 * @note Not therad-safe.
38 * @tparam Alloc Allocator type for internal storage (default:
39 * `std::allocator<std::byte>`)
40 */

270constexpr auto ConsumedMessagesRegistry<Alloc>::EnsureIds(
271 MessageTypeIndex type_index) -> ConsumedIds& {
272 const auto it = consumed_.lower_bound(type_index);
273 if (it != consumed_.end() && it->first == type_index) {
274 return it->second;

282template <typename Alloc>
283constexpr void ConsumedMessagesRegistry<Alloc>::MarkConsumed(
284 MessageTypeIndex type_index, AnyMessageId message_id) {
285 HELIOS_ASSERT(message_id.type == type_index,
286 "AnyMessageId type does not match type_index!");
287 auto& ids = EnsureIds(type_index);
288 // Insert in sorted order, maintaining uniqueness
289 const auto pos = std::ranges::lower_bound(ids, message_id);
290 if (pos == ids.end() || *pos != message_id) {
291 ids.insert(pos, message_id);
292 }
293}

295template <typename Alloc>
296template <typename OtherAlloc>
297constexpr void ConsumedMessagesRegistry<Alloc>::MergeFrom(
298 const ConsumedMessagesRegistry<OtherAlloc>& other) {
299 for (const auto& [type_index, other_indices] : other.Data()) {
300 if (other_indices.empty()) {
301 continue;
302 }
303
304 auto& our_ids = EnsureIds(type_index);
305 if (our_ids.empty()) {
306 our_ids.insert(our_ids.end(), other_indices.begin(), other_indices.end());
307 continue;
308 }
309
310 // In-place merge: append other ids, then inplace_merge, then deduplicate.
311 // This reuses existing capacity in our_ids, avoiding a separate allocation.
312 const auto original_size = our_ids.size();
313 our_ids.insert(our_ids.end(), other_indices.begin(), other_indices.end());
314 std::ranges::inplace_merge(
315 our_ids, our_ids.begin() + static_cast<ptrdiff_t>(original_size));
316 const auto [first, last] = std::ranges::unique(our_ids);
317 our_ids.erase(first, last);
318 }
319}

321template <typename Alloc>
322template <typename OtherAlloc>
323constexpr void ConsumedMessagesRegistry<Alloc>::MergeFrom(
324 ConsumedMessagesRegistry<OtherAlloc>&& other) {
325 for (auto&& [type_index, other_indices] : other.consumed_) {
326 if (other_indices.empty()) {
327 continue;
328 }
329
330 auto& our_ids = EnsureIds(type_index);
331 if (our_ids.empty()) {
332 our_ids = std::move(other_indices);
333 continue;
334 }
335
336 const auto original_size = our_ids.size();
337 our_ids.insert(our_ids.end(),
338 std::make_move_iterator(other_indices.begin()),
339 std::make_move_iterator(other_indices.end()));
340 std::ranges::inplace_merge(
341 our_ids, our_ids.begin() + static_cast<ptrdiff_t>(original_size));
342 const auto [first, last] = std::ranges::unique(our_ids);
343 our_ids.erase(first, last);
344 }
345
346 other.consumed_.clear();
347}
