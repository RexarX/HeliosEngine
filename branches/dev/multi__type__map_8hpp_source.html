27/**
28 * @brief Generic type-indexed map that stores one `Storage` instance per
29 * registered type key.
30 * @details Uses a `FlatMap` keyed by `TypeIndex` to store `Storage` instances.
31 * Each type gets its own storage entry identified by its compile-time type
32 * index. Type identification uses `helios::utils::TypeIndex`.
33 *
34 * This class models a *map structure* — use `Ensure<T>()` / `Get<T>()` to
35 * obtain the underlying `Storage` and interact with individual entries
36 * directly.
37 *
38 * If `Storage` supports `ChangeType<T>()`, it is called automatically on
39 * `Ensure<T>()`. If `Storage` supports `Merge(Storage&&)` or
40 * `merge(Storage&&)`, it is called during `Merge()` for overlapping type
41 * entries.
42 *
43 * @tparam Storage The value type stored per type key. Must be
44 * default-constructible.

687template <typename Storage>
688template <typename OtherStorage>
689constexpr void MultiTypeMap<Storage>::Merge(
690 const MultiTypeMap<OtherStorage>& other) {
691 for (const auto& [index, other_storage] : other.storage_) {
692 if (const auto it = storage_.Find(index); it != storage_.end()) {
693 // Existing key: prefer const overloads, then fall back to copy+move.
694 if constexpr (requires { it->second.Merge(other_storage); }) {
695 it->second.Merge(other_storage);
696 } else if constexpr (requires { it->second.merge(other_storage); }) {
697 it->second.merge(other_storage);
698 } else if constexpr (std::copy_constructible<OtherStorage> &&
699 requires(OtherStorage copy) {
700 it->second.Merge(std::move(copy));
701 }) {
702 auto copy = other_storage;
703 it->second.Merge(std::move(copy));
704 } else if constexpr (std::copy_constructible<OtherStorage> &&
705 requires(OtherStorage copy) {
706 it->second.merge(std::move(copy));
707 }) {
708 auto copy = other_storage;
709 it->second.merge(std::move(copy));
710 }
711 // else: no merge method available — existing entry is left unchanged.
712 } else {
713 // New key: insert storage.
714 if constexpr (std::same_as<Storage, OtherStorage> &&
715 std::copy_constructible<Storage>) {
716 // Same storage type — copy directly.
717 storage_.Emplace(index, other_storage);
718 } else {
719 // Different storage type — create a new default Storage and attempt to
720 // merge into it.
721 auto new_storage = MakeStorage();
722 if constexpr (requires { new_storage.Merge(other_storage); }) {
723 new_storage.Merge(other_storage);
724 } else if constexpr (requires { new_storage.merge(other_storage); }) {
725 new_storage.merge(other_storage);
726 } else if constexpr (std::copy_constructible<OtherStorage> &&
727 requires(OtherStorage copy) {
728 new_storage.Merge(std::move(copy));
729 }) {
730 auto copy = other_storage;
731 new_storage.Merge(std::move(copy));
732 } else if constexpr (std::copy_constructible<OtherStorage> &&
733 requires(OtherStorage copy) {
734 new_storage.merge(std::move(copy));
735 }) {
736 auto copy = other_storage;
737 new_storage.merge(std::move(copy));
738 } else if constexpr (std::constructible_from<Storage,
739 const OtherStorage&>) {
740 new_storage = Storage(other_storage);
741 }
742 storage_.Emplace(index, std::move(new_storage));
743 }
744 }
745 }

753 if (const auto it = storage_.Find(index); it != storage_.end()) {
754 // Existing key: call Merge or merge on Storage if available.
755 if constexpr (requires { it->second.Merge(std::move(other_storage)); }) {
756 it->second.Merge(std::move(other_storage));
757 } else if constexpr (requires {
758 it->second.merge(std::move(other_storage));
759 }) {
760 it->second.merge(std::move(other_storage));
761 }
762 // else: no merge method available — existing entry is left unchanged.
763 } else {
764 // New key: insert storage.
765 if constexpr (std::same_as<Storage, OtherStorage>) {
766 // Same storage type — move directly.
767 storage_.Emplace(index, std::move(other_storage));
768 } else {
769 // Different storage type — create a new default Storage and attempt to
770 // merge into it.
771 auto new_storage = MakeStorage();
772 if constexpr (requires {
773 new_storage.Merge(std::move(other_storage));
774 }) {
775 new_storage.Merge(std::move(other_storage));
776 } else if constexpr (requires {

778 }) {
779 new_storage.merge(std::move(other_storage));

781 new_storage = Storage(std::move(other_storage));
782 }
783 storage_.Emplace(index, std::move(new_storage));
784 }
785 }
786 }
