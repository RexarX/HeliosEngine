247template <typename T>
248constexpr auto FlattenValue(T&& value) {
249 using Decayed = std::remove_cvref_t<T>;
250 if constexpr (IsComponentBundleTypes<Decayed>::value) {
251 return FlattenBundleValues(std::forward<T>(value));
252 } else if constexpr (kIsNestedBundleType<Decayed>) {
253 if constexpr (std::is_lvalue_reference_v<T&&>) {
254 auto copy = value;
255 return FlattenBundleValues(std::move(copy).Build());
256 } else {
257 return FlattenBundleValues(std::move(value).Build());
258 }
259 } else {
260 return std::make_tuple(std::forward<T>(value));
