10/**
11 * @brief Semantic library / header version.
12 * @details `major` and `minor` are the compatibility key. `patch` is ignored
13 * by `Compatible`.
14 *
15 * Rules for header vs linked (DLL / shared object) versions:
16 * - Different `major` -> incompatible (ABI / API break).
17 * - Same `major`, linked `minor` < header `minor` -> incompatible (headers
18 * expect a newer binary).
19 * - Same `major`, linked `minor` >= header `minor` -> compatible (newer
20 * binary is fine).
21 */
22struct Version {
23 uint8_t major = 0;
24 uint8_t minor = 0;
25 uint16_t patch = 0;
26
27 [[nodiscard]] constexpr bool operator==(const Version&) const noexcept =
28 default;
29 [[nodiscard]] constexpr bool operator!=(const Version&) const noexcept =
30 default;
31 [[nodiscard]] constexpr auto operator<=>(const Version&) const noexcept =
32 default;
33};

61/**
62 * @brief Whether a binary at `linked` can satisfy headers at `header`.
63 * @param header Version of the headers the caller compiled against
64 * @param linked Version of the Helios binary being used
65 * @return `true` if majors match and `linked.minor >= header.minor`
66 */
67[[nodiscard]] constexpr bool Compatible(Version header,
68 Version linked) noexcept {
69 return linked.major == header.major && linked.minor >= header.minor;
70}

72/**

76[[nodiscard]] inline bool CompatibleWithHeaders() noexcept {
