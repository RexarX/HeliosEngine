50auto DynamicLibrary::Load(const std::filesystem::path& path)
51 -> DynamicLibraryResult<void> {
52 if (Loaded()) {

71 }

74 return {};

116auto DynamicLibrary::GetSymbolAddress(std::string_view name) const
