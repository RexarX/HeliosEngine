187/**
188 * @brief Concept for systems that declare all data access via parameter types.
189 * @details A system must be destructible, move or copy constructible, not
190 * polymorphic, an object type, and its `operator()` parameters must all have
191 * `SystemParamTraits` specialisations.
192 */
193template <typename T>
194concept SystemTrait =
195 std::destructible<T> &&
196 (std::move_constructible<T> || std::copy_constructible<T>) &&
197 !std::is_polymorphic_v<std::remove_cvref_t<T>> &&
198 std::is_object_v<std::remove_cvref_t<T>> &&
199 details::HasSystemParamCallOperator<T>;
200
201/**
202 * @brief Concept for lambda closure types that satisfy `SystemTrait`.
203 * @details Lambdas are detected via `utils::IsLambda<T>()`. They should always
204 * be registered with an explicit human-readable name.
205 */
206template <typename T>
207concept LambdaSystemTrait =
208 SystemTrait<T> && utils::LambdaTrait<std::remove_cvref_t<T>>;
