26/// @brief Global input behavior settings.
27/// @details Stick and trigger `AxisFilter`s apply to `Gamepad::axes`. Axis
28/// messages stay raw. `auto_calibrate` captures rest center on the first
29/// near-rest sample and recenters after `rest_frames` at rest.
30struct Settings {
31 static constexpr std::string_view kName = "helios::input::Settings";
32 static constexpr uint16_t kDefaultRestFrames = 60;
33
34 AxisFilter stick{.deadzone = AxisFilter::kDefaultDeadzone,
35 .livezone = AxisFilter::kDefaultLivezone,
36 .rescale = true};
37 AxisFilter trigger{.deadzone = AxisFilter::kDefaultTriggerDeadzone,
38 .livezone = AxisFilter::kDefaultLivezone,
39 .rescale = true};
40 uint16_t rest_frames = kDefaultRestFrames;
41 bool auto_calibrate = true;
42 bool raw_mouse_motion = false;
43};

45/**
46 * @brief Formats a input settings using an output iterator.
47 * @tparam It Output iterator type
48 * @param out Output iterator to write the formatted string to
49 * @param settings Input settings
50 * @return The output iterator after writing
51 */
52template <typename It>
53 requires std::output_iterator<It, char>
54inline It ToString(It out, const Settings& settings) {
55 out = std::format_to(out, "Settings{{stick=");
56 out = ToString(out, settings.stick);
57 out = std::format_to(out, ", trigger=");
58 out = ToString(out, settings.trigger);
59 return std::format_to(
60 out, ", rest_frames={}, auto_calibrate={}, raw_mouse_motion={}}}",
61 settings.rest_frames, settings.auto_calibrate, settings.raw_mouse_motion);
62}
