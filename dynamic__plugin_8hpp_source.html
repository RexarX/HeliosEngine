46 /**
47 * @brief Builds an export descriptor from a `PluginTypeId`.
48 * @param type_id Plugin type identity
49 * @return Export descriptor suitable for `helios_plugin_type_id`
50 */
51 [[nodiscard]] static constexpr PluginTypeExport From(
52 const PluginTypeId& type_id) noexcept {
53 const std::string_view qualified = type_id.QualifiedName();
54 const uint64_t hash = type_id.Index().Hash();
55 return {.hash = hash,
56 .qualified_name = qualified.empty() ? nullptr : qualified.data()};

144/**
145 * @brief Wrapper for dynamically loaded plugins.
146 * @details Loads a plugin from a shared library and manages its lifecycle.
147 * Supports hot-reloading: when the library file changes, the plugin can be
148 * unloaded and reloaded without restarting the application.
149 *
150 * The dynamic library must export:
151 * - A creation function (default: "helios_create_plugin") that returns
152 * `Plugin*`
153 * - A plugin type export function (default: "helios_plugin_type_id") that
154 * returns a `PluginTypeExport` via `PluginTypeExportFn`
155 *
156 * @note Not thread-safe.
157 * @code
158 * // In my_plugin.cpp (compiled as shared library)
159 * class MyPlugin : public helios::app::Plugin {
160 * public:
161 * static constexpr std::string_view kName = "MyPlugin";
162 *
163 * void Build(App& app) override { ... }
164 * void Destroy(App& app) override { ... }
165 * };
166 *
167 * extern "C" {
168 *
169 * HELIOS_EXPORT helios::app::Plugin* helios_create_plugin() {
170 * return new MyPlugin();
171 * }
172 *
173 * HELIOS_EXPORT auto helios_plugin_type_id()
174 * -> const helios::app::PluginTypeExport* {
175 * static const auto kExport =
176 * helios::app::PluginTypeExport::From<MyPlugin>();
177 * return &kExport;
178 * }
179 *
180 * }
181 * @endcode
182 *
183 * @code
184 * DynamicPlugin dyn_plugin;
185 * if (auto result = dyn_plugin.Load("my_plugin.so"); result) {
186 * app.AddDynamicPlugin(std::move(dyn_plugin));
187 * }
188 * @endcode
189 */
