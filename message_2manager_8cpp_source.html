76void MessageManager::ApplyConsumed(
77 const ConsumedMessagesRegistry& merged_consumed) {
78 HELIOS_ECS_PROFILE_SCOPE_N("helios::ecs::MessageManager::ApplyConsumed");
79 HELIOS_ECS_PROFILE_ZONE_VALUE(merged_consumed.TotalConsumedCount());
80
81 for (const auto& [type_index, consumed_ids] : merged_consumed.Data()) {
82 if (consumed_ids.empty()) {
83 continue;
84 }
85
86 const auto* metadata = registered_messages_.TryGet(type_index);
87 if (metadata == nullptr || metadata->is_async) {
88 continue;
89 }
90
91 auto* prev_ids = previous_ids_.TryGet(type_index);
92 auto* curr_ids = current_ids_.TryGet(type_index);
93 if ((prev_ids == nullptr || prev_ids->empty()) &&
94 (curr_ids == nullptr || curr_ids->empty())) {
95 continue;
96 }
97
98 std::pmr::vector<size_type> prev_indices{&mem::GetTemporaryStorage()};
99 std::pmr::vector<size_type> curr_indices{&mem::GetTemporaryStorage()};
100 prev_indices.reserve(consumed_ids.size());
101 curr_indices.reserve(consumed_ids.size());
102
103 for (const AnyMessageId id : consumed_ids) {
104 if (prev_ids != nullptr) {
105 const auto it = std::ranges::lower_bound(*prev_ids, id);
106 if (it != prev_ids->end() && *it == id) {
107 prev_indices.push_back(
108 static_cast<size_type>(it - prev_ids->begin()));
109 continue;
110 }
111 }
112 if (curr_ids != nullptr) {
113 const auto it = std::ranges::lower_bound(*curr_ids, id);
114 if (it != curr_ids->end() && *it == id) {
115 curr_indices.push_back(
116 static_cast<size_type>(it - curr_ids->begin()));
117 }
118 }
119 }
120
121 if (!prev_indices.empty()) {
122 previous_messages_.RemoveIndices(type_index, prev_indices);
123 RemoveIds(type_index, *prev_ids, prev_indices);
124 }
125 if (!curr_indices.empty()) {
126 current_messages_.RemoveIndices(type_index, curr_indices);
127 RemoveIds(type_index, *curr_ids, curr_indices);
128 }
129 }
130}

187void MessageManager::RemoveIds(MessageTypeIndex /*type_index*/,
188 MessageIdList& ids,
189 std::span<const size_type> sorted_indices) {
190 if (sorted_indices.empty()) {
191 return;
192 }
193
194 size_type idx = sorted_indices.size();
195 while (idx > 0) {
196 --idx;
197 size_type range_end = sorted_indices[idx] + 1;
198 size_type range_start = sorted_indices[idx];
199 while (idx > 0 && sorted_indices[idx - 1] == range_start - 1) {
200 --idx;
201 range_start = sorted_indices[idx];
202 }
203 ids.erase(ids.begin() + static_cast<ptrdiff_t>(range_start),
204 ids.begin() + static_cast<ptrdiff_t>(range_end));
205 }
