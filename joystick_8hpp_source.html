42/// @brief Snapshot of one unmapped joystick slot.
43struct Joystick {
44 static constexpr size_t kMaxAxes = 16;
45 static constexpr size_t kMaxButtons = 32;
46 static constexpr size_t kMaxHats = 4;
47
48 /// @brief Clears identity, counts, axes, buttons, and hats.
49 void Reset() noexcept;
50
51 std::string name;
52 std::string guid;
53 std::array<float, kMaxAxes> axes = {};
54 IndexedButtonInput<kMaxButtons> buttons;
55 std::optional<JoystickId> id;
56 std::array<JoystickHat, kMaxHats> hats = {};
57 uint8_t axis_count = 0;
58 uint8_t button_count = 0;
59 uint8_t hat_count = 0;
60 bool connected = false;
61};
62
63inline void Joystick::Reset() noexcept {
64 name.clear();
65 guid.clear();
66 axes.fill(0.0F);
67 buttons.Reset();
68 hats.fill(JoystickHat::kCentered);
69 axis_count = 0;
70 button_count = 0;
71 hat_count = 0;
72 id.reset();
73 connected = false;
74}

170/**
171 * @brief Returns the string name of a hat bitmask.
172 * @param hat Hat bits (named flags or combinations)
173 * @return String name of the hat
174 */
175[[nodiscard]] constexpr std::string_view ToString(JoystickHat hat) noexcept {
176 switch (static_cast<uint8_t>(hat)) {
177 case 0:
178 return "Centered";
179 case 1:
180 return "Up";
181 case 2:
182 return "Right";
183 case 3:
184 return "RightUp";
185 case 4:
186 return "Down";
187 case 6:
188 return "RightDown";
189 case 8:
190 return "Left";
191 case 9:
192 return "LeftUp";
193 case 12:
194 return "LeftDown";
195 default:
196 return "unknown";
197 }
198}
