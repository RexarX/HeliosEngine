93void ComponentManager::Clear(Entity entity) {
94 HELIOS_ASSERT(entity.Valid(), "Entity '{}' is invalid!", entity);
95 const auto it = entity_archetype_.find(entity.Index());
96 HELIOS_ASSERT(it != entity_archetype_.end(), "Entity '{}' not tracked!",
97 entity);
98
99 Archetype& current = it->second.get();
100
101 if (empty_archetype_ == nullptr) {
102 empty_archetype_ = &GetOrCreateArchetype(ArchetypeId{});
103 }
104
105 if (&current != empty_archetype_) {
106 Entity swapped = current.Remove(entity);
107 if (swapped.Valid()) {
108 entity_archetype_.insert_or_assign(swapped.Index(), std::ref(current));
109 }
110 empty_archetype_->AllocateRow(entity);
111 entity_archetype_.insert_or_assign(entity.Index(),
112 std::ref(*empty_archetype_));
113 ++structural_version_;
114 }
115
116 // Remove from all sparse storages.
117 for (auto& [type_index, entry] : sparse_storages_.Data()) {
118 entry.TryRemove(entity);

122Archetype& ComponentManager::GetOrCreateArchetype(const ArchetypeId& id) {
123 HELIOS_ECS_PROFILE_SCOPE_N(
124 "helios::ecs::ComponentManager::GetOrCreateArchetype");
125 HELIOS_ECS_PROFILE_ZONE_VALUE(id.Types().size());
126
127 const size_t hash = id.Hash();
128 if (const auto it = archetype_map_.find(hash); it != archetype_map_.end()) {
129 return it->second.archetype.get();
130 }
131
132 // Create archetype in the stable deque storage.
133 archetype_storage_.emplace_back(id, resource_);
134 Archetype& archetype = archetype_storage_.back();
135
136 // Create record referencing the archetype.
137 archetype_map_.emplace(hash, ArchetypeRecord(archetype, resource_));
138
139 // Initialize columns with type info from metadata.
140 InitArchetypeColumns(archetype);
141 archetype_list_.emplace_back(archetype);
142 ++structural_version_;
143 return archetype;
144}

160void ComponentManager::MigrateEntity(Entity entity, Archetype& src,
161 Archetype& dst) {
162 HELIOS_ECS_PROFILE_SCOPE_N("helios::ecs::ComponentManager::MigrateEntity");
163 HELIOS_ECS_PROFILE_ZONE_VALUE(src.EntityCount());
164 HELIOS_ECS_PROFILE_ZONE_VALUE(dst.Id().Types().size());
165
166 HELIOS_ASSERT(entity.Valid(), "Entity '{}' is invalid!", entity);
167
168 if (&src == &dst) [[unlikely]] {
169 return;
170 }
171
172 const auto src_row = src.Row(entity);
173
174 // Phase 1: Allocate a row in the destination archetype for this entity.
175 [[maybe_unused]] const auto dst_row = dst.AllocateRow(entity);
176
177 // Phase 2: For each column in dst, either move data from src or push a
178 // default placeholder.
179 for (const auto type_index : dst.Id().Types()) {
180 auto* dst_col = dst.TryColumn(type_index);
181 if (dst_col == nullptr) {
182 continue;
183 }
184
185 const auto meta_it = metadata_.Find(type_index);
186 HELIOS_ASSERT(meta_it != metadata_.end(),
187 "Component metadata not registered for a column in target "
188 "archetype!");
189
190 if (src.HasColumn(type_index)) {
191 // Shared column: move the element from src[src_row] into dst (appended
192 // at back).
193 auto* src_col = src.TryColumn(type_index);
194 if (src_col != nullptr && !src_col->Empty() &&
195 meta_it->second.move_column_element != nullptr) {
196 meta_it->second.move_column_element(*dst_col, *src_col,
197 static_cast<size_t>(src_row));
198 }
199 } else {
200 // New column in dst that src doesn't have.
201 // Push a default-constructed placeholder; the caller will overwrite it
202 // via Set.
203 if (meta_it->second.default_push != nullptr) {
204 meta_it->second.default_push(*dst_col);
205 }
206 }
207 }
208
209 // Phase 3: Remove the entity from the source archetype using swap-and-pop.
210 const Entity swapped = src.Remove(entity);
211
212 // Phase 4: Update entity-to-archetype mapping.
213 entity_archetype_.insert_or_assign(entity.Index(), std::ref(dst));
214 if (swapped.Valid()) {
215 entity_archetype_.insert_or_assign(swapped.Index(), std::ref(src));
216 }
217
218 ++structural_version_;
219}
