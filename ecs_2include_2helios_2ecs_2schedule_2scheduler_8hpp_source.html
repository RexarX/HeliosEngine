433inline auto StageOrdering::Before(this auto&& self, StageTypeIndex index)
434 -> decltype(std::forward<decltype(self)>(self)) {
435 const size_t other_hash = index.Hash();
436 auto& scheduler = self.scheduler_.get();
437 auto& entry = scheduler.GetStageEntry(self.Hash());
438
439 if (std::ranges::find(entry.before_stages, other_hash) ==
440 entry.before_stages.end()) {

442 }
443 scheduler.MarkDirty();
