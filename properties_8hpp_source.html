91/**
92 * @brief Combines dirty flags.
93 * @param lhs Left-hand side flag
94 * @param rhs Right-hand side flag
95 * @return Combined flag
96 */
97[[nodiscard]] constexpr DirtyFlag operator|(DirtyFlag lhs,

99 return static_cast<DirtyFlag>(std::to_underlying(lhs) |
100 std::to_underlying(rhs));

231/**
232 * @brief Computes a default client size for a monitor resolution.
233 * @details Uses most of the screen on portrait displays and a balanced fraction
234 * on landscape displays so fixed 1280x720 is not assumed.
235 * @param screen_width Monitor width in pixels
236 * @param screen_height Monitor height in pixels
237 * @return Default width and height in pixels
238 */
239[[nodiscard]] constexpr auto DefaultSizeForScreen(
240 uint32_t screen_width, uint32_t screen_height) noexcept
241 -> std::pair<uint32_t, uint32_t> {
242 constexpr uint32_t kMinSize = 320;
243
244 if (screen_width == 0 || screen_height == 0) {
245 return {1280, 720};
246 }
247
248 if (screen_height > screen_width) {
249 const uint32_t width =
250 std::clamp((screen_width * 9U) / 10U, kMinSize, screen_width);
251 const uint32_t height =
252 std::clamp((width * 16U) / 10U, kMinSize, (screen_height * 9U) / 10U);
253 return {width, height};
254 }
255
256 const uint32_t width =
257 std::clamp((screen_width * 2U) / 3U, kMinSize, screen_width);
258 const uint32_t height =
259 std::clamp((screen_height * 2U) / 3U, kMinSize, screen_height);
260 return {width, height};
261}

1009/**
1010 * @brief Formats dirty flags as a pipe-separated list and writes to an output
1011 * iterator.
1012 * @tparam It Output iterator type
1013 * @param out Output iterator to write the formatted string to
1014 * @param flags Combined dirty flags
1015 * @param with_prefix Whether to include a "DirtyFlag::" prefix for each flag
1016 * @return Updated output iterator after writing the formatted string
1017 */
1018template <typename It>
1019 requires std::output_iterator<It, char>
1020inline It ToString(It out, DirtyFlag flags, bool with_prefix = false) {
1021 const std::string_view kNoneStr = with_prefix ? "DirtyFlag::None" : "None";
1022
1023 if (flags == DirtyFlag::kNone) {
1024 return std::format_to(out, "{}", kNoneStr);
1025 }
1026
1027 bool first = true;
1028 const auto append = [flags, &out, &first, with_prefix](std::string_view name,
1029 DirtyFlag flag) {
1030 if (!HasFlag(flags, flag)) {
1031 return;
1032 }
1033 if (!first) {
1034 out = std::format_to(out, " | ");
1035 }
1036 first = false;
1037 out = with_prefix ? std::format_to(out, "DirtyFlag::{}", name)
1038 : std::format_to(out, "{}", name);
1039 };
1040
1041 append("Title", DirtyFlag::kTitle);
1042 append("Size", DirtyFlag::kSize);
1043 append("Pos", DirtyFlag::kPos);
1044 append("Mode", DirtyFlag::kMode);
1045 append("Cursor", DirtyFlag::kCursor);
1046 append("Visible", DirtyFlag::kVisible);
1047 append("Resizable", DirtyFlag::kResizable);
1048 append("Decorated", DirtyFlag::kDecorated);
1049 append("Monitor", DirtyFlag::kMonitor);
1050 append("Icon", DirtyFlag::kIcon);
1051 append("Maximized", DirtyFlag::kMaximized);
1052 append("SizeLimits", DirtyFlag::kSizeLimits);
1053 append("AspectRatio", DirtyFlag::kAspectRatio);
1054 append("Opacity", DirtyFlag::kOpacity);
1055 append("Floating", DirtyFlag::kFloating);
1056 append("AutoIconify", DirtyFlag::kAutoIconify);
1057 append("FocusOnShow", DirtyFlag::kFocusOnShow);
1058 append("Attention", DirtyFlag::kAttention);
1059 append("Focus", DirtyFlag::kFocus);
1060 append("MousePassthrough", DirtyFlag::kMousePassthrough);
1061 append("RefreshRate", DirtyFlag::kRefreshRate);
1062 return out;
1063}
