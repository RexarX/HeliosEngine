26struct FrameLimiterSettings {

41 /**
42 * @brief Manual cap at the given frames per second.
43 * @param fps Target frames per second (must be > 0)
44 * @return Manual settings with the corresponding interval

46 [[nodiscard]] static constexpr FrameLimiterSettings FromFPS(

48 constexpr size_t kNanosecsInSec = 1'000'000'000;
49 return {.mode = FrameLimiterMode::kManual,

53 /**
54 * @brief Manual cap at the given frequency in Hz.
55 * @param hz Target frequency in Hz (must be > 0)
56 * @return Manual settings with the corresponding interval

58 [[nodiscard]] static constexpr FrameLimiterSettings FromHz(
59 double hz) noexcept {
60 const auto nanosec = std::chrono::duration_cast<std::chrono::nanoseconds>(

65 /**
66 * @brief Manual cap with an explicit interval.
67 * @tparam Rep Duration representation type
68 * @tparam Period Duration period type
69 * @param interval Desired frame interval
70 * @return Manual settings with `interval`

72 template <typename Rep, typename Period>

75 return {.mode = FrameLimiterMode::kManual,
