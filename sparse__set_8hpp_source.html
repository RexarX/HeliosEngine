27/**
28 * @brief A sparse set data structure for efficient mapping of sparse indices to
29 * dense storage of values.
30 * @details SparseSet provides O(1) insertion, deletion, and lookup operations
31 * using two arrays:
32 * - sparse: maps element indices to dense indices
33 * - dense: stores packed values of type T in contiguous memory
34 *
35 * The data structure is particularly useful for managing sparse collections
36 * where indices may have large gaps but you want cache-friendly iteration over
37 * existing values.
38 *
39 * Memory complexity: O(max_index + num_elements)
40 * Time complexity: O(1) for all operations (amortized for insertions)
41 *
42 * @tparam T Type of values stored in the dense array
43 * @tparam IndexType Type used for element indices (default: size_t)
44 */

490template <typename T, typename IndexType>
491constexpr auto SparseSet<T, IndexType>::Insert(IndexType index, const T& value)
492 -> DenseIndexType {
493 HELIOS_ASSERT(IsValidIndex(index),
494 "Failed to insert value: index is invalid!");
495 if constexpr (std::is_signed_v<IndexType>) {
496 HELIOS_ASSERT(index >= 0,
497 "Failed to insert value: index cannot be negative!");
498 }
499
500 // Check if index already exists and replace the value
501 if (Contains(index)) {
502 const DenseIndexType dense_index = sparse_[index];
503 dense_[dense_index] = value;
504 return dense_index;
505 }
506
507 // Ensure sparse array can accommodate this index
508 if (index >= sparse_.size()) {
509 sparse_.resize(index + 1, kInvalidDenseIndex);
510 }
511
512 const auto dense_index = static_cast<DenseIndexType>(dense_.size());
513 dense_.push_back(value);
514 reverse_map_.push_back(index);
515 sparse_[index] = dense_index;
516
517 return dense_index;
518}

581template <typename T, typename IndexType>
582constexpr void SparseSet<T, IndexType>::Remove(IndexType index) noexcept {
583 HELIOS_ASSERT(IsValidIndex(index),
584 "Failed to remove value: index is invalid!");
585 if constexpr (std::is_signed_v<IndexType>) {
586 HELIOS_ASSERT(index >= 0,
587 "Failed to remove value: index cannot be negative!");
588 }
589 HELIOS_ASSERT(Contains(index),
590 "Failed to remove value: index does not exist!");
591
592 const DenseIndexType dense_index = sparse_[index];
593 const auto last_dense_index = static_cast<DenseIndexType>(dense_.size() - 1);
594
595 if (dense_index != last_dense_index) {
596 // Swap with last element to maintain dense packing
597 const IndexType last_index = reverse_map_[last_dense_index];
598 dense_[dense_index] = std::move(dense_[last_dense_index]);
599 reverse_map_[dense_index] = last_index;
600 sparse_[last_index] = dense_index;
601 }
602
603 dense_.pop_back();
604 reverse_map_.pop_back();
605 sparse_[index] = kInvalidDenseIndex;
606}
