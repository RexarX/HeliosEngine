714template <TypedBufferStorable T>
715 requires std::default_initializable<T>
716constexpr TypedBufferArray::TypedBufferArray(size_type count) {
717 if (count == 0) [[unlikely]] {
718 return;
719 }
720
721 type_info_ = TypeInfo::template From<T>();
722 storage_.resize(count * sizeof(T));
723
724 std::uninitialized_default_construct_n(DataPtr<T>(), count);
725 size_ = count;

1381constexpr void TypedBufferArray::Swap(size_type index, size_type other_index) {
1382 HELIOS_ASSERT(HasType(),
1383 "Cannot swap elements in storage without a type set!");
1384 HELIOS_ASSERT(index < size_, "Swap index '{}' is out of bounds!", index);
1385 HELIOS_ASSERT(other_index < size_, "Swap other_index '{}' is out of bounds!",
1386 other_index);
1387
1388 if (index == other_index) [[unlikely]] {
1389 return;
1390 }
1391
1392 auto* base = static_cast<std::byte*>(RawDataPtr());
1393 auto* first = base + (index * type_info_.element_size);
1394 auto* second = base + (other_index * type_info_.element_size);
1395
1396 if (type_info_.is_trivially_copyable) {
1397 std::swap_ranges(first, first + type_info_.element_size, second);
1398 } else {
1399 HELIOS_ASSERT(type_info_.move_construct,
1400 "Type must be move constructible to swap elements!");
1401
1402 std::pmr::polymorphic_allocator<std::byte> alloc(GetMemoryResource());
1403 using alloc_traits =
1404 std::allocator_traits<std::pmr::polymorphic_allocator<std::byte>>;
1405
1406 auto* temp = alloc_traits::allocate(alloc, type_info_.element_size);
1407
1408 try {
1409 type_info_.move_construct(temp, first, 1);
1410 if (type_info_.destroy != nullptr) {
1411 type_info_.destroy(first, 1);
1412 }
1413
1414 type_info_.move_construct(first, second, 1);
1415 if (type_info_.destroy != nullptr) {
1416 type_info_.destroy(second, 1);
1417 }
1418
