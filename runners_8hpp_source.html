27/// @brief Configuration for fixed-timestep runners.
28struct FixedRunnerConfig {
29 static constexpr auto kDefaultInterval =
30 std::chrono::nanoseconds{16'666'667}; // ~60 FPS default
31
32 /// Time to wait between the start of consecutive updates.

35 /**
36 * @brief Creates a config targeting the given frames per second.
37 * @param fps Target frames per second (must be > 0)
38 * @return FixedRunnerConfig with the corresponding update interval
39 */
40 [[nodiscard]] static constexpr auto FromFPS(size_t fps) noexcept
41 -> FixedRunnerConfig {
42 constexpr size_t kNanosecsInSec = 1'000'000'000;
43 return {.update_interval = std::chrono::nanoseconds{kNanosecsInSec / fps}};
44 }
45
46 /**
47 * @brief Creates a config targeting the given update frequency in Hz.
48 * @param hz Target frequency in Hz (must be > 0)
49 * @return FixedRunnerConfig with the corresponding update interval
50 */
51 [[nodiscard]] static constexpr FixedRunnerConfig FromHz(double hz) noexcept {
52 auto nanosec = std::chrono::duration_cast<std::chrono::nanoseconds>(

55 }

86/**
87 * @brief Runs the application with a fixed timestep.
88 * @details Uses `utils::PreciseSleepUntil` against an advancing absolute
89 * target time point so that scheduler wake-up overshoot on one frame is
90 * absorbed by a shorter sleep on the next, preventing drift accumulation.
91 * @param app Application to update
92 * @param config Fixed-runner configuration
93 * @return Exit code from the first `AppExit` message
94 */
95inline ExitCode RunFixed(App& app, const FixedRunnerConfig& config = {}) {
96 auto next_tick = std::chrono::steady_clock::now();
97
98 std::optional<ExitCode> exit_code;
99 while (exit_code = app.ShouldExit(), !exit_code) {
100 next_tick += config.update_interval;
101
102 app.Update();
103
104 utils::PreciseSleepUntil(next_tick);
105 // Prevent unbounded catch-up after a long stall (e.g. debugger pause)
106 next_tick = std::max(next_tick, std::chrono::steady_clock::now());
107 }
108 return *exit_code;
109}
