18 HELIOS_ASSERT(async_loops_running_.load(std::memory_order_acquire) == 0,

20 "Call Stop() or Shutdown() first.");

34 HELIOS_ASSERT(!blocking_update_future_.has_value(),

38 startup_graph_.Clear();
39 blocking_update_graph_.Clear();
40 shutdown_graph_.Clear();
41 sub_app_states_.clear();
42 blocking_update_future_.reset();

51void Scheduler::Build(App& app) {
52 HELIOS_APP_PROFILE_SCOPE_N("helios::app::Scheduler::Build");
53
54 auto& main = app.GetMainSubApp();
55 auto& executor = app.GetExecutor();
56
57 main.BuildScheduler(executor);
58
59 startup_graph_.Clear();
60 blocking_update_graph_.Clear();
61 shutdown_graph_.Clear();
62 sub_app_states_.clear();
63
64 if (app.sub_apps_.empty()) {
65 return;
66 }
67
68 sub_app_states_.reserve(app.sub_apps_.size());
69
70 for (auto&& [index, sub_app] : app.sub_apps_) {
71 sub_app.SetOwnerApp(app);
72 sub_app_states_.push_back(SubAppFrameState{
73 .sub_app = sub_app,
74 .mode = ClassifySubApp(sub_app),
75 });
76
77 startup_graph_.EmplaceTask(
78 [&sub_app, &executor]() { sub_app.RunStartupUnchecked(executor); });
79
80 shutdown_graph_.EmplaceTask([&sub_app, &executor]() {
81 sub_app.WaitUntilFullyIdle();
82 sub_app.RunShutdownUnchecked(executor);
83 });
84 }
85
86 for (auto&& [index, sub_app] : app.sub_apps_) {
87 if (ClassifySubApp(sub_app) != SubAppMode::kBlocking) {
88 continue;
89 }
90
91 blocking_update_graph_.EmplaceTask(
92 [&sub_app, &executor]() { sub_app.RunUpdatePass(executor); });
93 }

199void Scheduler::LaunchSubAppUpdates(App& app) {
200 if (app.sub_apps_.empty()) {
201 return;
202 }
203
204 auto& executor = app.GetExecutor();
205
206 if (blocking_update_graph_.TaskCount() > 0) {
207 blocking_update_future_ = executor.Run(blocking_update_graph_);
208 }
209
210 PruneCompletedOverlappingUpdates();
211
212 for (SubAppFrameState& state : sub_app_states_) {
213 if (state.mode != SubAppMode::kOverlapping ||
214 !state.fresh_extract_this_frame) {
215 continue;
216 }
217
218 SubApp& sub_app = state.sub_app.get();
219 overlapping_update_futures_.push_back(executor.Async(
220 [&sub_app, &executor]() { sub_app.RunUpdatePass(executor); }));

288void Scheduler::ExtractSubApp(SubAppFrameState& state,
289 const ecs::World& main_world) {
290 SubApp& sub_app = state.sub_app.get();
291 state.fresh_extract_this_frame = false;
292
293 switch (state.mode) {
294 using enum SubAppMode;
295 case kBlocking:
296 sub_app.WaitUntilFullyIdle();
297 sub_app.Extract(main_world);
298 state.fresh_extract_this_frame = true;
299 state.consecutive_extract_skips = 0;
300 return;
301 case kAsync:
302 sub_app.Extract(main_world, true);
303 return;
304 case kOverlapping:
305 break;
306 }
307
308 if (sub_app.IsUpdating()) {
309 const size_t max_skips = sub_app.MaxExtractionSkips();
310 if (max_skips == 0 || state.consecutive_extract_skips < max_skips) {
311 ++state.consecutive_extract_skips;
312 return;
313 }
314
315 sub_app.WaitUntilFullyIdle();
316 }
317
318 sub_app.Extract(main_world);
319 state.fresh_extract_this_frame = true;
320 state.consecutive_extract_skips = 0;
321}

323auto Scheduler::ClassifySubApp(const SubApp& sub_app) noexcept
324 -> Scheduler::SubAppMode {
325 if (sub_app.IsAsync()) {
326 return SubAppMode::kAsync;
327 }
328
329 if (sub_app.AllowsOverlappingUpdates()) {
330 return SubAppMode::kOverlapping;
331 }
332
333 return SubAppMode::kBlocking;
334}
