299constexpr void MessageQueue::Enqueue(T&& message) {
300 HELIOS_ASSERT(IsRegistered<std::remove_cvref_t<T>>(),
301 "Message type '{}' is not registered!",
302 MessageNameOf<std::remove_cvref_t<T>>());

304 .template EmplaceBack<std::remove_cvref_t<T>>(std::forward<T>(message));

316constexpr void MessageQueue::RemoveIndices(
317 MessageTypeIndex type_index, std::span<const size_type> sorted_indices) {
318 if (sorted_indices.empty()) [[likely]] {
319 return;
320 }
321
322 HELIOS_ASSERT(IsRegistered(type_index),
323 "Message type index is not registered!");
324
325 auto& buffer = messages_.Get(type_index);
326 if (buffer.Empty()) [[unlikely]] {
327 return;
328 }
329
330 // Erase contiguous ranges from back to front to preserve preceding indices.
331 auto idx = sorted_indices.size();
332 while (idx > 0) {
333 --idx;
334 auto range_end = sorted_indices[idx] + 1;
335 auto range_start = sorted_indices[idx];
336
337 // Coalesce consecutive indices into a single range.
338 while (idx > 0 && sorted_indices[idx - 1] == range_start - 1) {
339 --idx;
340 range_start = sorted_indices[idx];
341 }
342
343 buffer.Erase(range_start, range_end);
344 }
345}
